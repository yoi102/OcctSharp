/// <summary>Copied native workflow completion; unfinished results have no points.</summary>
public enum CurveComputationStatus {
 /// <summary>Completed with finite copied data.</summary>
 Complete=0,
 /// <summary>SDK did not complete or found no projection.</summary>
 NotDone=1,
 /// <summary>Native-only evaluation budget was exhausted.</summary>
 BudgetExceeded=2,
 /// <summary>Abscissa result lies outside the requested finite interval.</summary>
 OutsideDomain=3
}
/// <summary>Independent points and parameters. Length and Step are native diagnostics, not a proof of global approximation error.</summary>
public sealed record CurveSamplingResult(CurveComputationStatus Status,double Length,double Step,IReadOnlyList<Coordinates3d> Points,IReadOnlyList<double> Parameters) {
 /// <summary>All result points completed and were copied.</summary>
 public bool IsDone=>Status==CurveComputationStatus.Complete;
}
/// <summary>One detached orthogonal projection candidate with parameters and distance. Curve candidates use V=0.</summary>
public sealed record ProjectionCandidate(Coordinates3d Point,double U,double V,double Distance);
/// <summary>Copied SDK projection candidates in a finite domain. Equal nearest candidates remain ambiguous.</summary>
public sealed record BoundedProjectionResult(CurveComputationStatus Status,IReadOnlyList<ProjectionCandidate> Candidates,double MinimumDistance) {
 /// <summary>True when SDK completed and at least one finite candidate exists.</summary>
 public bool IsDone=>Status==CurveComputationStatus.Complete;
 /// <summary>Unique closest candidate index, or null on absence or equal-distance ambiguity within the given finite tolerance.</summary>
 public int? UniqueNearestIndex(double tolerance=1e-9){if(!double.IsFinite(tolerance)||tolerance<0)throw new ArgumentOutOfRangeException(nameof(tolerance));if(!IsDone)return null;int index=-1;for(int i=0;i<Candidates.Count;++i)if(Math.Abs(Candidates[i].Distance-MinimumDistance)<=tolerance){if(index>=0)return null;index=i;}return index>=0?index:null;}
}
