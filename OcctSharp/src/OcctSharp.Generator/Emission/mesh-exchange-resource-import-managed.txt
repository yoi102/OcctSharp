
 // The pinned OBJ reader resolves MTL textures against the OBJ folder and converts
 // texture names to ASCII. Normalize only explicit references in a private copy;
 // source identities/relative graph and all geometry/material values remain exact.
 static (string Main,Dictionary<string,string> Names) ObjSdkGraph(string stage,string main,string[] names,List<FileStream> locks)
 {
  string folder=Path.Combine(stage,"sdk-obj");Directory.CreateDirectory(folder);
  var forward=new Dictionary<string,string>(StringComparer.OrdinalIgnoreCase);
  var reverse=new Dictionary<string,string>(StringComparer.OrdinalIgnoreCase);
  for(int n=0;n<names.Length;++n){string alias="r"+n.ToString("D3",System.Globalization.CultureInfo.InvariantCulture)+Path.GetExtension(names[n]);forward.Add(names[n],alias);reverse.Add("sdk-obj/"+alias,names[n]);}
  var materials=new HashSet<string>(StringComparer.OrdinalIgnoreCase);
  foreach(string line in File.ReadLines(Path.Combine(stage,main)))if(line.Trim().StartsWith("mtllib ",StringComparison.Ordinal))materials.Add(ResourceName(line.Trim()[7..].Trim()));
  foreach(string name in names)
  {
   string source=Path.Combine(stage,name),target=Path.Combine(folder,forward[name]);
   if(name.Equals(main,StringComparison.OrdinalIgnoreCase))
   {
    using var output=new StreamWriter(target,false,ProjectionCodec.Utf8);
    foreach(string line in File.ReadLines(source)){string trimmed=line.Trim();output.WriteLine(trimmed.StartsWith("mtllib ",StringComparison.Ordinal)?"mtllib "+forward[ResourceName(trimmed[7..].Trim())]:line);}
   }
   else if(materials.Contains(name))
   {
    using var output=new StreamWriter(target,false,ProjectionCodec.Utf8);
    foreach(string line in File.ReadLines(source))
    {
     string row=line.Trim();int space=row.IndexOfAny([' ','\t']);string keyword=space<0?row:row[..space];
     if(space<0||!keyword.StartsWith("map_",StringComparison.OrdinalIgnoreCase)&&!keyword.Equals("bump",StringComparison.OrdinalIgnoreCase)){output.WriteLine(line);continue;}
     string local=ResourceName(row[(space+1)..].Trim());
     string original=ResourceName(Path.GetRelativePath(stage,Path.GetFullPath(local,Path.GetDirectoryName(source)!)).Replace('\\','/'));
     output.WriteLine(keyword+" "+forward[original]);
    }
   }
   else File.Copy(source,target,false);
   locks.Add(new FileStream(target,FileMode.Open,FileAccess.Read,FileShare.Read));
  }
  return(Path.Combine(folder,forward[main]),reverse);
 }

 static void GltfResourceIndices(JsonElement root)
 {
  static JsonElement[] Rows(JsonElement parent,string key,int limit)
  {
   if(!parent.TryGetProperty(key,out var array))return [];
   if(array.ValueKind!=JsonValueKind.Array||array.GetArrayLength()>limit)throw new InvalidDataException("Resource table budget/type: "+key);
   return array.EnumerateArray().ToArray();
  }
  static int Index(JsonElement value,int count)
  {
   if(!value.TryGetInt32(out int index)||index<0||index>=count)throw new InvalidDataException("Resource index is outside its exact array.");
   return index;
  }
  var images=Rows(root,"images",128);var views=Rows(root,"bufferViews",65536);
  foreach(var image in images)
  {
   bool uri=image.TryGetProperty("uri",out _),view=image.TryGetProperty("bufferView",out var bufferView);
   if(uri==view)throw new InvalidDataException("Image requires one exact resource source.");
   if(view)Index(bufferView,views.Length);
   if(image.TryGetProperty("mimeType",out var mime)&&mime.GetString()!="image/png")throw new NotSupportedException("Only PNG resource textures are admitted.");
   if(view&&!image.TryGetProperty("mimeType",out _))throw new InvalidDataException("Buffer-view image requires its PNG MIME type.");
  }
  var samplers=Rows(root,"samplers",128);var textures=Rows(root,"textures",128);
  foreach(var texture in textures){Index(texture.GetProperty("source"),images.Length);if(texture.TryGetProperty("sampler",out var sampler))Index(sampler,samplers.Length);}
  void Texture(JsonElement parent,string key)
  {
   if(!parent.TryGetProperty(key,out var value))return;
   Index(value.GetProperty("index"),textures.Length);
   if(value.TryGetProperty("texCoord",out var coordinate)&&(!coordinate.TryGetInt32(out int channel)||channel!=0))throw new NotSupportedException("Only the first imported texture coordinate channel is admitted.");
  }
  var materials=Rows(root,"materials",4096);
  foreach(var material in materials)
  {
   if(material.TryGetProperty("pbrMetallicRoughness",out var pbr)){Texture(pbr,"baseColorTexture");Texture(pbr,"metallicRoughnessTexture");}
   Texture(material,"normalTexture");Texture(material,"occlusionTexture");Texture(material,"emissiveTexture");
  }
  foreach(var mesh in Rows(root,"meshes",4096))foreach(var primitive in Rows(mesh,"primitives",100000))if(primitive.TryGetProperty("material",out var material))Index(material,materials.Length);
 }

  else{foreach(string line in File.ReadLines(main)){string text=line.Trim();if(!text.StartsWith("mtllib ",StringComparison.Ordinal))continue;string name=ResourceName(text[7..].Trim());Add(name);string material=Path.GetFullPath(name,parent);NoReparse(material);if(new FileInfo(material).Length>67108864)throw new InvalidDataException("MTL byte budget.");foreach(string mtlLine in File.ReadLines(material)){string row=mtlLine.Trim();int space=row.IndexOf(' ');if(space<0)continue;string keyword=row[..space];if(!keyword.StartsWith("map_",StringComparison.OrdinalIgnoreCase)&&!keyword.Equals("bump",StringComparison.OrdinalIgnoreCase))continue;string texture=ResourceName(row[(space+1)..].Trim());Add(Path.GetRelativePath(parent,Path.GetFullPath(texture,Path.GetDirectoryName(material)!)).Replace('\\','/'));}}}
  return names.Order(StringComparer.OrdinalIgnoreCase).ToArray();}

 /// <summary>Copies encoded PNG bytes and decoded corner colors from one native material-owned texture. The snapshot survives document disposal.</summary>
 public static unsafe MeshTextureSnapshot? ReadTexture(DocumentProjectionContext document,DocumentProjectionLabel material,MeshTextureSlot slot)
 {
  ArgumentNullException.ThrowIfNull(document);document.Validate(material);if(!Enum.IsDefined(slot))throw new ArgumentOutOfRangeException(nameof(slot));Ready();
  NativeError.ThrowIfFailed(TextureNative(document.Parent,document.Domain,material.Entry,(int)slot,out nint token),"mesh_texture_snapshot");
  using var payload=new ProjectionPayloadHandle(token,2);using var reader=ProjectionCodec.Reader(payload,2);
  if(reader.ReadInt32()!=83)throw new InvalidDataException("Wrong mesh texture snapshot.");
  if(!ProjectionCodec.Boolean(reader)){ProjectionCodec.End(reader);return null;}
  int width=reader.ReadInt32(),height=reader.ReadInt32(),size=ProjectionCodec.Count(reader,8388608);
  if(width<=0||height<=0||(long)width*height>16777216)throw new InvalidDataException("Texture dimensions outside copied profile.");
  byte[] bytes=reader.ReadBytes(size);if(bytes.Length!=size)throw new InvalidDataException("Truncated encoded PNG snapshot.");
  int count=ProjectionCodec.Count(reader,4);var corners=new MeshTexturePixel[count];
  for(int n=0;n<count;++n)corners[n]=new(reader.ReadInt32(),reader.ReadInt32(),reader.ReadDouble(),reader.ReadDouble(),reader.ReadDouble(),reader.ReadDouble());
  ProjectionCodec.End(reader);return new(slot,width,height,Array.AsReadOnly(bytes),Array.AsReadOnly(corners));
 }
