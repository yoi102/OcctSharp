 }

  auto color=occ::down_cast<XCAFDoc_Color>(target);OcctSharpDocumentProjections::Require(!color.IsNull(),"Invalid exact XML color target.");color->Set(Quantity_ColorRGBA(static_cast<float>(values[0]),static_cast<float>(values[1]),static_cast<float>(values[2]),static_cast<float>(values[3])));return true;
 }
};
