constexpr int HatchLimit=1048576;
static_assert(TopAbs_IN==0&&TopAbs_OUT==1&&TopAbs_ON==2&&TopAbs_UNKNOWN==3,"Copied hatching region state ordinals changed.");

void HatchFinite(double x){Require(std::isfinite(x),"Hatching parameters must be finite.");}

opencascade::handle<Geom2d_Curve> HatchCopy(const void* token,int kind) {
 opencascade::handle<Geom2d_Curve> value;
 switch(kind) {
 case 0:value=OcctSharpGenerated::ValidateGeom2dCurve(static_cast<const OcctSharp_Geom2dCurveHandle*>(token))->Value;break;
 case 1:value=OcctSharpGenerated::ValidateGeom2dLine(static_cast<const OcctSharp_Geom2dLineHandle*>(token))->Value;break;
 case 2:value=OcctSharpGenerated::ValidateGeom2dCircle(static_cast<const OcctSharp_Geom2dCircleHandle*>(token))->Value;break;
 case 3:value=OcctSharpGenerated::ValidateGeom2dBSplineCurve(static_cast<const OcctSharp_Geom2dBSplineCurveHandle*>(token))->Value;break;
 case 4:value=OcctSharpGenerated::ValidateGeom2dBezierCurve(static_cast<const OcctSharp_Geom2dBezierCurveHandle*>(token))->Value;break;
 case 5:value=OcctSharpGenerated::ValidateGeom2dTrimmedCurve(static_cast<const OcctSharp_Geom2dTrimmedCurveHandle*>(token))->Value;break;
 default:Require(false,"Invalid hatching curve kind.");
 }
 auto copy=opencascade::handle<Geom2d_Curve>::DownCast(value->Copy());Require(!copy.IsNull(),"Hatching curve copy failed.");return copy;
}
HatchEventData HatchEvent(const HatchGen_IntersectionPoint& point,int type) {
 HatchEventData result;
 result.parameter=point.Parameter();HatchFinite(result.parameter);
 result.fields={point.Index(),point.Position(),point.StateBefore(),point.StateAfter(),point.SegmentBeginning(),point.SegmentEnd(),type};return result;
}
const HatchResultData& HatchCurveAt(const HatchSnapshotData& snapshot,int index) {
 Require(index>=0&&static_cast<size_t>(index)<snapshot.curves.size(),"Hatching index out of range.");return snapshot.curves[index];
}
const HatchDomainData& HatchDomainAt(const HatchResultData& curve,int index) {
 Require(curve.info[0]&&!curve.info[1]&&curve.info[2]&&curve.info[3]==HatchGen_NoProblem,"Hatching domains did not succeed.");Require(index>=0&&static_cast<size_t>(index)<curve.domains.size(),"Hatching domain index out of range.");return curve.domains[index];
}

void OCCTSHARP_CALL occtsharp_hatching_release(void* token) { OcctSharpOwned::Release(token,typeid(HatchSnapshotData)); }
OcctSharp_Status OCCTSHARP_CALL occtsharp_hatching_create(const void* const* inputs,const int32_t* kinds,const double* ranges,int32_t boundaryCount,int32_t hatchCount,const int32_t* orientations,const double* tolerances,int32_t flags,void** result,int32_t* done,double* actualTolerances,int32_t capacity) {
 if(!result||!done)return OCCTSHARP_STATUS_INVALID_ARGUMENT;*result=nullptr;*done=0;
 return Guard([&]{
  Require(boundaryCount>=0&&hatchCount>=0&&int64_t(boundaryCount)+hatchCount<=HatchLimit,"Invalid hatching input counts.");
  Require(int64_t(boundaryCount)*hatchCount<=HatchLimit,"Too many boundary/hatching pairs.");
  Require(tolerances&&actualTolerances&&capacity>=4&&flags>=0&&flags<=3,"Invalid hatching options or output buffer.");
  int count=boundaryCount+hatchCount;Require(count==0||(inputs&&kinds&&ranges),"Missing hatching inputs.");Require(boundaryCount==0||orientations,"Missing boundary orientation.");
  for(int i=0;i<4;++i){HatchFinite(tolerances[i]);Require(tolerances[i]>0,"Hatching tolerances must be positive.");}
  Geom2dHatch_Intersector intersector(tolerances[0],tolerances[1]);
  Geom2dHatch_Hatcher hatcher(intersector,tolerances[2],tolerances[3],(flags&1)!=0,(flags&2)!=0);
  auto snapshot=std::make_shared<HatchSnapshotData>();snapshot->tolerances={intersector.ConfusionTolerance(),intersector.TangencyTolerance(),hatcher.Confusion2d(),hatcher.Confusion3d()};snapshot->keepPoints=hatcher.KeepPoints();snapshot->keepSegments=hatcher.KeepSegments();
  std::vector<opencascade::handle<Geom2d_Curve>> copies;copies.reserve(count);
  std::vector<int> indices;indices.reserve(hatchCount);
  for(int i=0;i<count;++i){
   double first=ranges[i*2],last=ranges[i*2+1];HatchFinite(first);HatchFinite(last);Require(first<last&&std::isfinite(last-first),"Invalid finite hatching range.");auto copy=HatchCopy(inputs[i],kinds[i]);
   if(!copy->IsPeriodic())Require(first>=copy->FirstParameter()&&last<=copy->LastParameter(),"Range exceeds nonperiodic curve domain.");
   Geom2dAdaptor_Curve adaptor(copy,first,last);copies.push_back(copy);
   if(i<boundaryCount){Require(orientations[i]==TopAbs_FORWARD||orientations[i]==TopAbs_REVERSED,"Boundary must be forward or reversed.");hatcher.AddElement(adaptor,static_cast<TopAbs_Orientation>(orientations[i]));}
   else indices.push_back(hatcher.AddHatching(adaptor));
  }
  hatcher.Trim();hatcher.ComputeDomains();bool complete=true;size_t expanded=0;
  for(int i=0;i<hatchCount;++i){
   int index=indices[i];HatchResultData curve;curve.curve=copies[boundaryCount+i];
   curve.info={hatcher.TrimDone(index),hatcher.TrimFailed(index),hatcher.IsDone(index),hatcher.Status(index),0,0};
   complete=complete&&curve.info[0]&&!curve.info[1]&&curve.info[2]&&curve.info[3]==HatchGen_NoProblem;
   if(curve.info[0]&&!curve.info[1]){
    int points=hatcher.NbPoints(index);Require(points>=0,"Invalid native point count.");expanded+=points;HatchSize(expanded);
    for(int p=1;p<=points;++p){const auto& point=hatcher.Point(index,p);HatchPointData output;output.event=HatchEvent(point,-1);int elements=point.NbPoints();Require(elements>=0,"Invalid native element count.");expanded+=elements;HatchSize(expanded);
     for(int e=1;e<=elements;++e){const auto& element=point.Point(e);output.elements.push_back(HatchEvent(element,element.IntersectionType()));}curve.points.push_back(std::move(output));}
   }
   if(curve.info[0]&&!curve.info[1]&&curve.info[2]&&curve.info[3]==HatchGen_NoProblem){
    int domains=hatcher.NbDomains(index);Require(domains>=0,"Invalid native domain count.");expanded+=domains;HatchSize(expanded);
    for(int d=1;d<=domains;++d){const auto& domain=hatcher.Domain(index,d);HatchDomainData output;output.hasFirst=domain.HasFirstPoint();output.hasLast=domain.HasSecondPoint();
     if(output.hasFirst){output.first=domain.FirstPoint().Parameter();HatchFinite(output.first);}if(output.hasLast){output.last=domain.SecondPoint().Parameter();HatchFinite(output.last);}
     output.clippedFirst=output.hasFirst?std::max(output.first,ranges[(boundaryCount+i)*2]):ranges[(boundaryCount+i)*2];
     output.clippedLast=output.hasLast?std::min(output.last,ranges[(boundaryCount+i)*2+1]):ranges[(boundaryCount+i)*2+1];curve.domains.push_back(output);}
   }
   curve.info[4]=static_cast<int>(curve.points.size());curve.info[5]=static_cast<int>(curve.domains.size());snapshot->curves.push_back(std::move(curve));
  }
  Publication<HatchSnapshotData> publication(snapshot);std::copy(snapshot->tolerances.begin(),snapshot->tolerances.end(),actualTolerances);*done=complete;*result=publication.Commit();
 });
}

  Require(pointIndex>=0&&static_cast<size_t>(pointIndex)<curve.points.size(),"Hatching point index out of range.");const auto& point=curve.points[pointIndex];Require(elementIndex>=-1&&(elementIndex==-1||static_cast<size_t>(elementIndex)<point.elements.size()),"Hatching element index out of range.");const auto& event=elementIndex==-1?point.event:point.elements[elementIndex];std::copy(event.fields.begin(),event.fields.end(),values);*parameter=event.parameter;*elementCount=static_cast<int>(point.elements.size());});

OcctSharp_Status OCCTSHARP_CALL occtsharp_hatching_domain(const void* token,int32_t index,int32_t domainIndex,int32_t* flags,int32_t flagCapacity,double* values,int32_t capacity) {

  const int32_t f[]={domain.hasFirst,domain.hasLast,domain.clippedFirst<domain.clippedLast};const double v[]={domain.first,domain.last,domain.clippedFirst,domain.clippedLast};std::copy(f,f+3,flags);std::copy(v,v+4,values);});
}
OcctSharp_Status OCCTSHARP_CALL occtsharp_hatching_trimmed(const void* token,int32_t index,int32_t domainIndex,int32_t dimension,OcctSharp_Value_Plane plane,void** result) {
 if(!result)return OCCTSHARP_STATUS_INVALID_ARGUMENT;*result=nullptr;
 return Guard([&]{Require(dimension==2||dimension==3,"Invalid trimmed curve dimension.");auto snapshot=OcctSharpOwned::Lease<const HatchSnapshotData>(token);const auto& curve=HatchCurveAt(*snapshot,index);const auto& domain=HatchDomainAt(curve,domainIndex);
  if(domain.clippedFirst>=domain.clippedLast)return;
  auto copied=opencascade::handle<Geom2d_Curve>::DownCast(curve.curve->Copy());Require(!copied.IsNull(),"Trimmed curve copy failed.");
  opencascade::handle<Geom2d_Curve> trimmed=new Geom2d_TrimmedCurve(copied,domain.clippedFirst,domain.clippedLast,true,false);
  if(dimension==2)*result=OcctSharpGenerated::AllocateGeom2dCurve(trimmed);
  else{auto lifted=GeomAPI::To3d(trimmed,OcctSharp_GeometryValues::ToNative(plane));Require(!lifted.IsNull(),"Planar curve lift failed.");*result=OcctSharpGenerated::AllocateGeomCurve(lifted);}
 });
