using OcctSharp.Interop;

namespace OcctSharp;

public sealed partial class OcctViewer
{
    private readonly Dictionary<long, ViewerSource> _sources = [];
    private static void RequireSourceRuntime()
    { var abi=OcctRuntime.Info.AbiVersion; if(abi.Major!=1 || abi.Minor<117) throw new NotSupportedException("Viewer sources require ABI1.117 or later."); }

    /// <summary>
    /// Copies the shape's topology into this viewer while retaining exact
    /// original identities for selection. The caller may then release the shape.
    /// </summary>
    public ViewerSource CreateSource(Shape shape)
    {
        ArgumentNullException.ThrowIfNull(shape);
        EnsureThread(); RequireSourceRuntime();
        NativeError.ThrowIfFailed(
            NativeMethods.CreateViewerShapeSource(Handle, shape.Handle, out long id),
            "viewer_source_shape");
        return RegisterSource(id);
    }

    /// <summary>
    /// Copies an XDE label's topology and effective styles into this viewer.
    /// The caller may then release the source document.
    /// </summary>
    public ViewerSource CreateSource(XdeLabel label)
    {
        ArgumentNullException.ThrowIfNull(label);
        EnsureThread(); RequireSourceRuntime();
        NativeError.ThrowIfFailed(
            NativeMethods.CreateViewerXdeSource(Handle, label.Document.Handle, label.Entry, out long id),
            "viewer_source_xde");
        return RegisterSource(id);
    }

    /// <summary>
    /// Displays an instance at a copied transform. The logical presentation and
    /// selection owner remain independent of other instances of this source.
    /// </summary>
    public ViewerPresentation Display(ViewerSource source, GpTrsf transform,
        ViewerSourceIdentity? identity = null)
    {
        ArgumentNullException.ThrowIfNull(source);
        ArgumentNullException.ThrowIfNull(transform);
        EnsureThread(); RequireSourceRuntime();
        if (!ReferenceEquals(source.Viewer, this))
            throw new ArgumentException("The source belongs to another viewer.", nameof(source));
        ObjectDisposedException.ThrowIf(source.IsRemoved, source);
        NativeError.ThrowIfFailed(
            NativeMethods.DisplayViewerSource(Handle, source.Id, transform.Handle, out long id),
            "viewer_source_display");
        ViewerPresentation presentation = new(this, id, identity);
        _presentations.Add(id, presentation);
        if (_isolationVisibility is not null)
        {
            _isolationVisibility[id] = true;
            SetVisible(presentation, false);
        }
        return presentation;
    }

    /// <summary>Releases every source creation entry, retaining existing instances.</summary>
    public void ClearSources()
    {
        EnsureThread(); RequireSourceRuntime();
        foreach (ViewerSource source in _sources.Values.ToArray()) RemoveSource(source);
    }

    private ViewerSource RegisterSource(long id)
    {
        ViewerSource source = new(this, id);
        _sources.Add(id, source);
        return source;
    }

    internal void RefreshSourceStyles(ViewerSource source, XdeLabel label)
    {
        ArgumentNullException.ThrowIfNull(label);
        EnsureThread(); RequireSourceRuntime();
        ObjectDisposedException.ThrowIf(source.IsRemoved, source);
        NativeError.ThrowIfFailed(
            NativeMethods.RefreshViewerXdeSource(Handle, source.Id, label.Document.Handle, label.Entry),
            "viewer_source_refresh_xde");
    }

    internal void RemoveSource(ViewerSource source)
    {
        if (source.IsRemoved) return;
        EnsureThread(); RequireSourceRuntime();
        if (!ReferenceEquals(source.Viewer, this))
            throw new ArgumentException("The source belongs to another viewer.", nameof(source));
        NativeError.ThrowIfFailed(NativeMethods.ReleaseViewerSource(Handle, source.Id), "viewer_source_release");
        _sources.Remove(source.Id);
        source.MarkRemoved();
    }
}
