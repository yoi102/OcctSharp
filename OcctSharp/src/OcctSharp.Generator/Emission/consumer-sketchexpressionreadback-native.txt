constexpr int Limit=1000000;
void Require(bool condition,const char* message) {
  if(!condition)throw OperationFailure(OCCTSHARP_STATUS_INVALID_ARGUMENT,message);
}
struct Span {std::uintptr_t first,last;};
Span Bytes(const void* pointer,int count,std::size_t size) {
  Require(count>=0&&count<=Limit,"Snapshot span exceeds its resource bound.");
  if(count==0)return{0,0};
  Require(pointer,"Snapshot span pointer is null.");
  const auto first=reinterpret_cast<std::uintptr_t>(pointer);
  const auto bytes=static_cast<std::size_t>(count)*size;
  Require(first<=std::numeric_limits<std::uintptr_t>::max()-bytes,"Snapshot span overflow.");
  return{first,first+bytes};
}
void Disjoint(Span a,Span b) {
  Require(!a.first||!b.first||a.last<=b.first||b.last<=a.first,"Snapshot outputs alias another span.");
}
void Bound(double first,double last) {
  Require(std::isfinite(first)&&std::isfinite(last)&&first<last,"Snapshot curve has invalid parameter bounds.");
}
gp_Dir2d Direction(OcctSharp_SketchPoint2d value) {
  auto point=ToSketchPoint(value,"Snapshot direction is not finite.");
  Require(std::abs(std::hypot(point.X(),point.Y())-1.0)<=1e-10,"Snapshot direction must be a unit vector.");
  return gp_Dir2d(point.X(),point.Y());
}
OcctSharp_SketchPoint2d CopyDirection(const gp_Dir2d& d){return{d.X(),d.Y()};}
struct Snapshot {
  std::vector<OcctSharp_SketchNativeNode> nodes;
  std::vector<OcctSharp_SketchPoint2d> poles;
  std::vector<double> weights,knots;
  std::vector<int32_t> multiplicities;
  int Add(const opencascade::handle<Geom_Curve>& curve,const gp_Pln& plane,
    double edge_first,double edge_last,double tolerance,int depth) {
    Require(!curve.IsNull()&&depth<64,"Snapshot native tree exceeds64nodes or has a null basis.");
    OcctSharp_SketchNativeNode node{};node.basis_index=-1;node.flags=curve->IsPeriodic()?2:0;
    node.first_parameter=curve->FirstParameter();node.last_parameter=curve->LastParameter();
    Bound(node.first_parameter,node.last_parameter);
    if(const auto trim=opencascade::handle<Geom_TrimmedCurve>::DownCast(curve);!trim.IsNull()) {
      node.kind=6;node.basis_index=Add(trim->BasisCurve(),plane,edge_first,edge_last,tolerance,depth+1);
    } else if(const auto offset=opencascade::handle<Geom_OffsetCurve>::DownCast(curve);!offset.IsNull()) {
      Require(offset->Direction().IsParallel(plane.Axis().Direction(),1e-10),"Snapshot offset direction leaves the sketch plane normal.");
      node.kind=7;node.basis_index=Add(offset->BasisCurve(),plane,edge_first,edge_last,tolerance,depth+1);
      node.signed_offset=offset->Offset()*(offset->Direction().Dot(plane.Axis().Direction())<0?-1.0:1.0);
      Require(std::isfinite(node.signed_offset),"Snapshot offset distance is not finite.");
    } else {
      // Project only primitives. Traversing the actual3D wrappers above avoids
      // OCCT constructor simplification changing the tree we are reporting.
      const auto value=ProjectExactSketchCurve(curve,plane,edge_first,edge_last,tolerance,0);
      if(const auto line=opencascade::handle<Geom2d_Line>::DownCast(value);!line.IsNull()) {
        node.kind=1;node.origin=FromSketchPoint(line->Location());node.x_direction=CopyDirection(line->Direction());
      } else if(const auto circle=opencascade::handle<Geom2d_Circle>::DownCast(value);!circle.IsNull()) {
        node.kind=2;const auto position=circle->Position();node.origin=FromSketchPoint(position.Location());
        node.x_direction=CopyDirection(position.XDirection());node.y_direction=CopyDirection(position.YDirection());
        node.major_radius=circle->Radius();node.minor_radius=circle->Radius();
      } else if(const auto ellipse=opencascade::handle<Geom2d_Ellipse>::DownCast(value);!ellipse.IsNull()) {
        node.kind=3;const auto position=ellipse->Position();node.origin=FromSketchPoint(position.Location());
        node.x_direction=CopyDirection(position.XDirection());node.y_direction=CopyDirection(position.YDirection());
        node.major_radius=ellipse->MajorRadius();node.minor_radius=ellipse->MinorRadius();
      } else if(const auto bezier=opencascade::handle<Geom2d_BezierCurve>::DownCast(value);!bezier.IsNull()) {
        node.kind=4;node.degree=bezier->Degree();node.flags=bezier->IsRational()?1:0;
        node.pole_start=static_cast<int>(poles.size());node.pole_count=bezier->NbPoles();
        Require(node.pole_count<=Limit-node.pole_start,"Snapshot pole count exceeds its resource bound.");
        for(int i=1;i<=node.pole_count;++i){poles.push_back(FromSketchPoint(bezier->Pole(i)));weights.push_back(bezier->Weight(i));}
      } else if(const auto spline=opencascade::handle<Geom2d_BSplineCurve>::DownCast(value);!spline.IsNull()) {
        node.kind=5;node.degree=spline->Degree();node.flags=(spline->IsRational()?1:0)|(spline->IsPeriodic()?2:0);
        node.pole_start=static_cast<int>(poles.size());node.pole_count=spline->NbPoles();
        node.knot_start=static_cast<int>(knots.size());node.knot_count=spline->NbKnots();
        Require(node.pole_count<=Limit-node.pole_start&&node.knot_count<=Limit-node.knot_start,"Snapshot table count exceeds its resource bound.");
        for(int i=1;i<=node.pole_count;++i){poles.push_back(FromSketchPoint(spline->Pole(i)));weights.push_back(spline->Weight(i));}
        for(int i=1;i<=node.knot_count;++i){knots.push_back(spline->Knot(i));multiplicities.push_back(spline->Multiplicity(i));}
      } else throw OperationFailure(OCCTSHARP_STATUS_TYPE_MISMATCH,"Unsupported exact native sketch primitive.");
    }
    nodes.push_back(node);return static_cast<int>(nodes.size())-1;
  }
};
auto Build(const OcctSharp_SketchNativeSummary& summary,const OcctSharp_SketchNativeNode* nodes,
  const OcctSharp_SketchPoint2d* poles,const double* weights,const double* knots,const int32_t* multiplicities) {
  std::vector<opencascade::handle<Geom2d_Curve>> values;values.reserve(static_cast<std::size_t>(summary.node_count));
  for(int i=0;i<summary.node_count;++i) {
    const auto& n=nodes[i];Bound(n.first_parameter,n.last_parameter);
    Require(n.kind>=1&&n.kind<=7&&n.flags>=0&&n.flags<=3,"Snapshot node kind or flags are invalid.");
    Require(n.pole_start>=0&&n.pole_count>=0&&n.pole_start<=summary.pole_count&&n.pole_count<=summary.pole_count-n.pole_start&&
      n.knot_start>=0&&n.knot_count>=0&&n.knot_start<=summary.knot_count&&n.knot_count<=summary.knot_count-n.knot_start,
      "Snapshot table index is out of range.");
    opencascade::handle<Geom2d_Curve> value;
    if(n.kind==6||n.kind==7) {
      Require(n.basis_index==i-1&&i>0&&n.pole_count==0&&n.knot_count==0&&!(n.flags&1),"Snapshot wrapper basis is invalid.");
      const auto& basis=values[static_cast<std::size_t>(n.basis_index)];
      if(n.kind==6)value=new Geom2d_TrimmedCurve(basis,n.first_parameter,n.last_parameter,true,false);
      else {Require(std::isfinite(n.signed_offset),"Snapshot offset distance is not finite.");value=new Geom2d_OffsetCurve(basis,n.signed_offset,true);}
    } else {
      Require(i==0&&n.basis_index==-1,"Snapshot primitive must be the unique first node.");
      if(n.kind<=3) {
        Require(n.flags==(n.kind==1?0:2)&&n.pole_count==0&&n.knot_count==0,"Snapshot analytic node has unexpected tables/periodicity.");
        auto origin=ToSketchPoint(n.origin,"Snapshot analytic origin is not finite.");auto x=Direction(n.x_direction);
        if(n.kind==1)value=new Geom2d_Line(origin,x);
        else {
          const auto y=Direction(n.y_direction);
          Require(std::abs(x.Dot(y))<=1e-10,"Snapshot conic axes must be orthogonal.");
          Require(std::isfinite(n.major_radius)&&std::isfinite(n.minor_radius)&&n.minor_radius>0&&n.major_radius>=n.minor_radius,
            "Snapshot conic radii are invalid.");
          const gp_Ax22d axes(origin,x,y);
          if(n.kind==2){Require(n.major_radius==n.minor_radius,"Snapshot circle radii differ.");value=new Geom2d_Circle(axes,n.major_radius);}
          else value=new Geom2d_Ellipse(axes,n.major_radius,n.minor_radius);
        }
      } else {
        Require(n.pole_count>=2&&n.degree>0&&n.degree<=Geom2d_BSplineCurve::MaxDegree(),"Snapshot freeform degree/poles are invalid.");
        NCollection_Array1<gp_Pnt2d> points(1,n.pole_count);NCollection_Array1<double> w(1,n.pole_count);
        for(int j=0;j<n.pole_count;++j) {
          points.SetValue(j+1,ToSketchPoint(poles[n.pole_start+j],"Snapshot control point is not finite."));
          const auto weight=weights[n.pole_start+j];Require(std::isfinite(weight)&&weight>0,"Snapshot control weight is invalid.");w.SetValue(j+1,weight);
          Require((n.flags&1)||weight==1,"Snapshot non-rational weight must be one.");
        }
        if(n.kind==4) {
          Require(n.degree==n.pole_count-1&&!(n.flags&2)&&n.knot_count==0,"Snapshot Bezier metadata is inconsistent.");
          value=(n.flags&1)?new Geom2d_BezierCurve(points,w):new Geom2d_BezierCurve(points);
        } else {
          Require(n.knot_count>=2,"Snapshot B-spline has too few knots.");
          NCollection_Array1<double> k(1,n.knot_count);NCollection_Array1<int> m(1,n.knot_count);
          for(int j=0;j<n.knot_count;++j) {
            const auto knot=knots[n.knot_start+j];const auto mult=multiplicities[n.knot_start+j];
            Require(std::isfinite(knot)&&mult>0&&mult<=n.degree+1&&(j==0||knot>knots[n.knot_start+j-1]),"Snapshot knot table is invalid.");
            k.SetValue(j+1,knot);m.SetValue(j+1,mult);
          }
          value=(n.flags&1)?new Geom2d_BSplineCurve(points,w,k,m,n.degree,(n.flags&2)!=0):
            new Geom2d_BSplineCurve(points,k,m,n.degree,(n.flags&2)!=0);
        }
      }
    }
    Require(value->IsPeriodic()==((n.flags&2)!=0),"Snapshot periodic metadata is inconsistent with its curve.");
    values.push_back(value);
  }
  return values.back();
}

OcctSharp_Status OCCTSHARP_CALL occtsharp_sketch_expression_read_edge(
  const OcctSharp_ShapeHandle* shape,const OcctSharp_SketchPlane* plane,double tolerance,
  OcctSharp_SketchNativeSummary* summary,OcctSharp_SketchNativeNode* nodes,int32_t node_capacity,
  OcctSharp_SketchPoint2d* poles,double* weights,int32_t pole_capacity,double* knots,int32_t* multiplicities,int32_t knot_capacity) {
  return Guard([&] {
    Require(plane&&summary&&std::isfinite(tolerance)&&tolerance>0,"Invalid snapshot plane,output or tolerance.");
    ValidateUsableShape(shape);
    Require(shape->Value.ShapeType()==TopAbs_EDGE,"Snapshot requires a single edge.");
    Require(shape->Value.Orientation()==TopAbs_FORWARD||shape->Value.Orientation()==TopAbs_REVERSED,"Snapshot cannot represent internal/external edge traversal.");
    Require(node_capacity>=0&&node_capacity<=64&&pole_capacity>=0&&pole_capacity<=Limit&&knot_capacity>=0&&knot_capacity<=Limit,
      "Snapshot output capacity exceeds its resource bound.");
    const bool query=!nodes&&!poles&&!weights&&!knots&&!multiplicities&&node_capacity==0&&pole_capacity==0&&knot_capacity==0;
    Require(query||(nodes&&node_capacity>0&&(pole_capacity==0?(!poles&&!weights):(poles&&weights))&&
      (knot_capacity==0?(!knots&&!multiplicities):(knots&&multiplicities))),"Snapshot output buffers must be filled together.");
    const std::vector<Span> outputs={Bytes(summary,1,sizeof(*summary)),Bytes(nodes,node_capacity,sizeof(*nodes)),
      Bytes(poles,pole_capacity,sizeof(*poles)),Bytes(weights,pole_capacity,sizeof(*weights)),
      Bytes(knots,knot_capacity,sizeof(*knots)),Bytes(multiplicities,knot_capacity,sizeof(*multiplicities))};
    for(std::size_t i=0;i<outputs.size();++i) {
      Disjoint(outputs[i],Bytes(plane,1,sizeof(*plane)));Disjoint(outputs[i],Bytes(shape,1,sizeof(*shape)));
      for(std::size_t j=0;j<i;++j)Disjoint(outputs[i],outputs[j]);
    }
    const auto edge=TopoDS::Edge(shape->Value);double first,last;const auto curve=BRep_Tool::Curve(edge,first,last);Bound(first,last);
    Snapshot snapshot;snapshot.Add(curve,BuildSketchPlane(*plane),first,last,tolerance,0);
    const OcctSharp_SketchNativeSummary result={static_cast<int>(snapshot.nodes.size()),static_cast<int>(snapshot.poles.size()),
      static_cast<int>(snapshot.knots.size()),edge.Orientation()==TopAbs_REVERSED?1:0,first,last};
    if(!query) {
      Require(result.node_count<=node_capacity&&result.pole_count<=pole_capacity&&result.knot_count<=knot_capacity,"Snapshot output capacity is insufficient.");
      std::copy(snapshot.nodes.begin(),snapshot.nodes.end(),nodes);
      if(result.pole_count){std::copy(snapshot.poles.begin(),snapshot.poles.end(),poles);std::copy(snapshot.weights.begin(),snapshot.weights.end(),weights);}
      if(result.knot_count){std::copy(snapshot.knots.begin(),snapshot.knots.end(),knots);std::copy(snapshot.multiplicities.begin(),snapshot.multiplicities.end(),multiplicities);}
    }
    *summary=result;
  });
}
OcctSharp_Status OCCTSHARP_CALL occtsharp_sketch_expression_snapshot_evaluate(
  const OcctSharp_SketchNativeSummary* summary,const OcctSharp_SketchNativeNode* nodes,
  const OcctSharp_SketchPoint2d* poles,const double* weights,const double* knots,const int32_t* multiplicities,
  double parameter,int32_t order,OcctSharp_SketchEvaluation* output) {
  return Guard([&] {
    Require(summary&&output,"Snapshot evaluation output/input is null.");
    Require(summary->node_count>0&&summary->node_count<=64&&summary->pole_count>=0&&summary->pole_count<=Limit&&
      summary->knot_count>=0&&summary->knot_count<=Limit&&(summary->reversed==0||summary->reversed==1),"Snapshot summary exceeds its resource bound or has invalid direction.");
    const auto destination=Bytes(output,1,sizeof(*output));
    for(const auto& span:std::vector<Span>{Bytes(summary,1,sizeof(*summary)),Bytes(nodes,summary->node_count,sizeof(*nodes)),
      Bytes(poles,summary->pole_count,sizeof(*poles)),Bytes(weights,summary->pole_count,sizeof(*weights)),
      Bytes(knots,summary->knot_count,sizeof(*knots)),Bytes(multiplicities,summary->knot_count,sizeof(*multiplicities))})Disjoint(span,destination);
    Bound(summary->first_parameter,summary->last_parameter);
    Require(std::isfinite(parameter)&&parameter>=summary->first_parameter&&parameter<=summary->last_parameter&&order>=1&&order<=3,
      "Snapshot evaluation parameter or derivative order is out of range.");
    const auto curve=Build(*summary,nodes,poles,weights,knots,multiplicities);
    Require(curve->IsPeriodic()||(summary->first_parameter>=nodes[summary->node_count-1].first_parameter&&
      summary->last_parameter<=nodes[summary->node_count-1].last_parameter),"Snapshot edge interval leaves its native root.");
    const auto native=summary->reversed?summary->first_parameter+(summary->last_parameter-parameter):parameter;
    const auto point=curve->Value(native);auto derivative=ExactSketchDerivative2d(curve,native,order);
    if(summary->reversed&&order%2)derivative.Reverse();
    Require(std::isfinite(point.X())&&std::isfinite(point.Y())&&std::isfinite(derivative.X())&&std::isfinite(derivative.Y()),"Snapshot evaluation is not finite.");
    const OcctSharp_SketchEvaluation result={FromSketchPoint(point),{derivative.X(),derivative.Y()},parameter};*output=result;
  });
}
