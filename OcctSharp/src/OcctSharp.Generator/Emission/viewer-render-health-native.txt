#include "OcctSharp.Native.Internal.hxx"
#include "OcctSharp.ViewerRenderingHealth.Generated.h"
#include "Runtime/Error.hxx"
#include "Visualization/Context.hxx"
#include <Graphic3d_StructureManager.hxx>
#include <OpenGl_Context.hxx>
#include <OpenGl_GraphicDriver.hxx>
#include <V3d_Viewer.hxx>

using namespace OcctSharp::Native;

OcctSharp_Status OCCTSHARP_CALL occtsharp_viewer_render_health(
  OcctSharp_ViewerHandle* viewer, int32_t* output)
{
  if (output == nullptr)
  {
    SetLastError("The rendering health output pointer is null.");
    return OCCTSHARP_STATUS_INVALID_ARGUMENT;
  }
  *output = 0;
  return Guard([&]
  {
    ValidateViewerThread(viewer);
    int32_t flags = viewer->Viewer->StructureManager()->IsDeviceLost() ? 1 : 0;
    const auto& context = viewer->Driver->GetSharedContext();
    if (context.IsNull()) flags |= 2;
    else if (!context->IsValid()) flags |= 4;
    // Publish only after successful validation and reads; no native object escapes.
    *output = flags;
  });
}

OcctSharp_Status OCCTSHARP_CALL occtsharp_viewer_invalidate_gpu_resources(
  OcctSharp_ViewerHandle* viewer)
{
  return Guard([&]
  {
    ValidateViewerThread(viewer);
    viewer->Viewer->StructureManager()->SetDeviceLost();
  });
}
