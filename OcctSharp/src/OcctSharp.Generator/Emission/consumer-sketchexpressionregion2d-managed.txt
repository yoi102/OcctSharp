namespace OcctSharp;

#pragma warning disable CS1591

/// <summary>A copied strict containment tree for exact closed expression loops.
/// Even-depth contours are filled; their immediate odd-depth children are holes.</summary>
public sealed class SketchExpressionRegion2d
{
    private readonly SketchExpressionChain2d[] loops;
    private readonly int[] parents, depths;
    private readonly SketchExpressionProfile2d[] profiles;
    private SketchExpressionRegion2d(SketchExpressionChain2d[] loops, int[] parents, int[] depths,
        SketchExpressionProfile2d[] profiles, double tolerance) =>
        (this.loops, this.parents, this.depths, this.profiles, Tolerance) = (loops, parents, depths, profiles, tolerance);
    public const int MaximumLoopCount = 256;
    public IReadOnlyList<SketchExpressionChain2d> Loops => Array.AsReadOnly((SketchExpressionChain2d[])loops.Clone());
    public IReadOnlyList<int> ParentIndices => Array.AsReadOnly((int[])parents.Clone());
    public IReadOnlyList<int> Depths => Array.AsReadOnly((int[])depths.Clone());
    public IReadOnlyList<SketchExpressionProfile2d> Profiles => Array.AsReadOnly((SketchExpressionProfile2d[])profiles.Clone());
    public double Tolerance { get; }

    public static SketchExpressionRegion2d Classify(IReadOnlyList<SketchExpressionChain2d> loops, double tolerance = 1e-7)
    {
        ArgumentNullException.ThrowIfNull(loops);
        SketchModeling.ValidateTolerance(tolerance, nameof(tolerance));
        if (loops.Count is < 1 or > MaximumLoopCount) throw new ArgumentException("A region requires 1 through 256 loops.", nameof(loops));
        if (loops.Any(value => value is null)) throw new ArgumentException("An expression loop is null.", nameof(loops));
        if (loops.Sum(value => value.Curves.Count) > SketchExpressionChain2d.MaximumCurveCount)
            throw new ArgumentException("A region exceeds the 4096-curve resource bound.", nameof(loops));
        return ClassifyChecked(loops.Select(value => SketchExpressionChain2d.CreateOrdered(value.Curves, true, tolerance)).ToArray(), tolerance);
    }

    internal static SketchExpressionRegion2d ClassifyChecked(SketchExpressionChain2d[] loops, double tolerance)
    {
        int count = loops.Length;
        if (count is < 1 or > MaximumLoopCount || loops.Sum(value => value.Curves.Count) > SketchExpressionChain2d.MaximumCurveCount)
            throw new ArgumentException("An expression region exceeds its loop or curve resource bound.", nameof(loops));
        var contains = new bool[count, count];
        var areas = loops.Select(value => value.Measure().SignedArea).ToArray();
        for (int i = 0; i < count; ++i)
            if (!double.IsFinite(areas[i]) || Math.Abs(areas[i]) <= tolerance * tolerance)
                throw new SketchValidationException([new(SketchDiagnosticCode.ZeroLength, $"Expression loop {i} encloses no finite area.", i)]);
        for (int i = 0; i < count; ++i)
            for (int j = i + 1; j < count; ++j)
            {
                foreach (var a in loops[i].Curves)
                    foreach (var b in loops[j].Curves)
                    {
                        var intersection = a.InspectIntersections(b, tolerance);
                        if (intersection.HasOverlappingIntervals || intersection.Endpoints.Count != 0)
                            throw new SketchValidationException([new(SketchDiagnosticCode.AmbiguousNesting,
                                $"Expression loops {i} and {j} intersect, touch or overlap.", i, j)]);
                    }
                contains[i,j] = loops[i].Contains(loops[j].Curves[0].Evaluate(0).Point, tolerance);
                contains[j,i] = loops[j].Contains(loops[i].Curves[0].Evaluate(0).Point, tolerance);
                if (contains[i,j] && contains[j,i])
                    throw new SketchValidationException([new(SketchDiagnosticCode.AmbiguousNesting,
                        $"Expression loops {i} and {j} have ambiguous mutual containment.", i, j)]);
            }
        var depth = Enumerable.Range(0, count).Select(j => Enumerable.Range(0, count).Count(i => contains[i,j])).ToArray();
        var parents = Enumerable.Repeat(-1, count).ToArray();
        for (int j = 0; j < count; ++j)
        {
            var candidates = Enumerable.Range(0, count).Where(i => contains[i,j] && depth[i] == depth[j] - 1).ToArray();
            if (depth[j] == 0) continue;
            if (candidates.Length != 1)
                throw new SketchValidationException([new(SketchDiagnosticCode.AmbiguousNesting,
                    $"Expression loop {j} has no unique immediate parent.", j)]);
            parents[j] = candidates[0];
        }
        for (int j = 0; j < count; ++j)
        {
            var ancestors = new HashSet<int>();
            for (int p = parents[j]; p >= 0; p = parents[p])
                if (!ancestors.Add(p)) throw new SketchValidationException([new(SketchDiagnosticCode.AmbiguousNesting, "Expression containment has a cycle.", j, p)]);
            if (!ancestors.SetEquals(Enumerable.Range(0, count).Where(i => contains[i,j])))
                throw new SketchValidationException([new(SketchDiagnosticCode.AmbiguousNesting, $"Expression loop {j} has inconsistent containment.", j)]);
        }
        var normalized = loops.Select((loop,i) => (areas[i] > 0) == (depth[i] % 2 == 0) ? loop : loop.Reverse()).ToArray();
        var profiles = Enumerable.Range(0, count).Where(i => depth[i] % 2 == 0)
            .Select(i => SketchExpressionProfile2d.FromClassified(normalized[i],
                Enumerable.Range(0,count).Where(j => parents[j] == i).Select(j => normalized[j]).ToArray(), tolerance)).ToArray();
        return new(normalized, parents, depth, profiles, tolerance);
    }

    public Shape[] CreateFaces(SketchPlane plane)
    {
        ArgumentNullException.ThrowIfNull(plane);
        List<Shape> result = [];
        try
        {
            foreach (var profile in profiles) result.Add(profile.CreateFace(plane));
            return result.ToArray();
        }
        catch { foreach (var face in result) face.Dispose(); throw; }
    }

    public Shape CreateShape(SketchPlane plane)
    {
        var faces = CreateFaces(plane);
        try { return ShapeFactory.CreateCompound(faces); }
        finally { foreach (var face in faces) face.Dispose(); }
    }
}

#pragma warning restore CS1591
