using OcctSharp.Interop;

namespace OcctSharp;

#pragma warning disable CS1591

/// <summary>An immutable, explicitly ordered and directed exact expression
/// chain. Construction validates native contacts without reordering, reversing,
/// fitting or dropping author curves.</summary>
public sealed class SketchExpressionChain2d
{
    public const int MaximumCurveCount = 4096;
    private readonly SketchCurveExpression2d[] curves;
    private SketchExpressionChain2d(SketchCurveExpression2d[] curves, bool closed, double tolerance) =>
        (this.curves, IsClosed, Tolerance) = (curves, closed, tolerance);
    public IReadOnlyList<SketchCurveExpression2d> Curves => Array.AsReadOnly((SketchCurveExpression2d[])curves.Clone());
    public bool IsClosed { get; }
    public double Tolerance { get; }

    public static SketchExpressionChain2d CreateOrdered(
        IReadOnlyList<SketchCurveExpression2d> curves, bool requireClosed = false, double tolerance = 1e-7)
    {
        ArgumentNullException.ThrowIfNull(curves);
        SketchModeling.ValidateTolerance(tolerance, nameof(tolerance));
        if (curves.Count is < 1 or > MaximumCurveCount)
            throw new ArgumentException("An expression chain requires 1 through 4096 curves.", nameof(curves));
        var source = curves.Select(value => value ?? throw new ArgumentException("An expression curve is null.", nameof(curves))).ToArray();
        List<SketchDiagnostic> diagnostics = [];
        var starts = new SketchPoint2d[source.Length];
        var ends = new SketchPoint2d[source.Length];
        for (int i = 0; i < source.Length; ++i)
        {
            starts[i] = source[i].Evaluate(0).Point;
            ends[i] = source[i].Evaluate(1).Point;
            using var edge = source[i].ToEdge(SketchPlane.XY);
            if (edge.GetEdgeLength() <= tolerance)
                diagnostics.Add(new(SketchDiagnosticCode.ZeroLength, $"Expression curve {i} has zero length.", i));
            var self = source[i].InspectSelfIntersections(tolerance);
            if (self.HasOverlappingIntervals || self.Endpoints.Any(value => !IsPeriodicSeam(value)))
                diagnostics.Add(new(SketchDiagnosticCode.SelfIntersection, $"Expression curve {i} intersects itself.", i, i));
        }
        for (int i = 1; i < source.Length; ++i)
            if (ends[i - 1].DistanceTo(starts[i]) > tolerance)
                diagnostics.Add(new(SketchDiagnosticCode.Gap, $"Expression curves {i - 1} and {i} have a gap in the supplied order.", i - 1, i));
        bool closed = ends[^1].DistanceTo(starts[0]) <= tolerance;
        if (requireClosed && !closed)
            diagnostics.Add(new(SketchDiagnosticCode.OpenChain, "The explicitly ordered expression chain is open."));
        if (diagnostics.Count != 0) throw new SketchValidationException(diagnostics);
        for (int i = 0; i < source.Length; ++i)
            for (int j = i + 1; j < source.Length; ++j)
            {
                var intersections = source[i].InspectIntersections(source[j], tolerance);
                if (intersections.HasOverlappingIntervals)
                {
                    diagnostics.Add(new(SketchDiagnosticCode.DuplicateCurve, $"Expression curves {i} and {j} overlap.", i, j));
                    continue;
                }
                if (intersections.Endpoints.Any(value => !AllowedJoint(value, i, j, source.Length, closed)))
                    diagnostics.Add(new(SketchDiagnosticCode.SelfIntersection, $"Expression curves {i} and {j} intersect away from their supplied joint.", i, j));
            }
        if (diagnostics.Count != 0) throw new SketchValidationException(diagnostics);
        return new(source, closed, tolerance);
    }

    /// <summary>Reverses the entire directed chain while retaining its exact
    /// geometry and tolerance. No native topology identity is inferred.</summary>
    public SketchExpressionChain2d Reverse() =>
        new(curves.Reverse().Select(value => value.Reverse()).ToArray(), IsClosed, Tolerance);

    public unsafe Shape BuildWire(SketchPlane plane)
    {
        ArgumentNullException.ThrowIfNull(plane);
        List<Shape> edges = new(curves.Length);
        try
        {
            foreach (var curve in curves) edges.Add(curve.ToEdge(plane));
            return ShapeFactory.WithBorrowedShapeHandles(edges, (handles, count) =>
            {
                NativeError.ThrowIfFailed(NativeMethods.SketchMakeWire(handles, count, Tolerance, out nint wire), "sketch_expression_make_wire");
                return ShapeFactory.FromNativeHandle(wire, "sketch_expression_make_wire");
            });
        }
        finally { foreach (var edge in edges) edge.Dispose(); }
    }

    public SketchLoopMeasurement Measure()
    {
        using var wire = BuildWire(SketchPlane.XY);
        var bounds = wire.GetBoundingBox();
        double area = 0, compensation = 0;
        var origin = curves[0].Evaluate(0).Point;
        if (IsClosed)
            foreach (var curve in curves)
            {
                double value = SketchExpressionArea.Integrate(curve, origin) - compensation;
                double next = area + value;
                compensation = (next - area) - value;
                area = next;
            }
        return new(wire.InspectProperties(InspectionPropertyKind.Length).Mass, area, area > 0,
            new(new(bounds.Minimum.X, bounds.Minimum.Y), new(bounds.Maximum.X, bounds.Maximum.Y)));
    }

    internal bool Contains(SketchPoint2d point, double tolerance)
    {
        SketchCurveExpression2d.RequireRuntime(78);
        using var wire = BuildWire(SketchPlane.XY);
        NativeError.ThrowIfFailed(NativeMethods.SketchExpressionWireContains(wire.Handle, new(point.X, point.Y),
            tolerance, out int inside), "sketch_expression_wire_contains");
        return inside != 0;
    }

    private static bool IsPeriodicSeam(SketchIntersection value) =>
        Math.Min(value.FirstNormalizedParameter, value.SecondNormalizedParameter) <= 1e-7 &&
        Math.Max(value.FirstNormalizedParameter, value.SecondNormalizedParameter) >= 1 - 1e-7;
    private static bool AllowedJoint(SketchIntersection value, int first, int second, int count, bool closed) =>
        second == first + 1 && Math.Abs(value.FirstNormalizedParameter - 1) <= 1e-7 && Math.Abs(value.SecondNormalizedParameter) <= 1e-7 ||
        closed && first == 0 && second == count - 1 && Math.Abs(value.FirstNormalizedParameter) <= 1e-7 && Math.Abs(value.SecondNormalizedParameter - 1) <= 1e-7;
}

#pragma warning restore CS1591
