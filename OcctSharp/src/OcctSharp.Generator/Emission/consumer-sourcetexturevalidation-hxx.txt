#pragma once
#include "OcctSharp.ConsumerSourceTextures.Generated.hxx"
#include <Image_CompressedPixMap.hxx>
#include <Image_PixMap.hxx>
#include <Image_SupportedFormats.hxx>
#include <unordered_set>

namespace OcctSharp::Native
{
using ValidatedViewerTextures = std::unordered_set<const Image_Texture*>;

inline void ValidateViewerTexture(const opencascade::handle<Image_Texture>& texture,
  const opencascade::handle<Image_SupportedFormats>& formats, ValidatedViewerTextures& validated)
{
  if (texture.IsNull() || validated.contains(texture.get())) return;
  // Validate the detached encoded snapshot. Keep GPU-compressed data encoded;
  // an inability to decode DDS on the CPU must not reject a readable DDS image.
  const auto compressed = texture->ReadCompressedImage(formats);
  if (!compressed.IsNull() && compressed->SizeX() > 0 && compressed->SizeY() > 0
      && !compressed->FaceData().IsNull() && !compressed->FaceData()->IsEmpty())
  {
    validated.insert(texture.get());
    return;
  }
  const auto& encoded = texture->DataBuffer();
  // FreeImage can produce pixels from a truncated DXT stream after OCCT's
  // compressed reader rejects it. Do not treat that permissive fallback as
  // proof that the detached GPU-compressed payload is complete.
  if (!encoded.IsNull() && encoded->Size() >= 88
      && std::memcmp(encoded->Data(), "DDS ", 4) == 0
      && (std::memcmp(encoded->Data() + 84, "DXT1", 4) == 0
          || std::memcmp(encoded->Data() + 84, "DXT3", 4) == 0
          || std::memcmp(encoded->Data() + 84, "DXT5", 4) == 0))
    throw OperationFailure(OCCTSHARP_STATUS_INVALID_ARGUMENT,
      "Source texture has invalid or truncated compressed image data.");
  const auto image = texture->ReadImage(formats);
  if (image.IsNull() || image->IsEmpty())
    throw OperationFailure(OCCTSHARP_STATUS_INVALID_ARGUMENT,
      "Source texture has invalid or unsupported encoded image data.");
  validated.insert(texture.get());
}

inline void ValidateViewerMaterialTextures(const opencascade::handle<XCAFDoc_VisMaterial>& material,
  ValidatedViewerTextures& validated)
{
  opencascade::handle<Image_SupportedFormats> formats = new Image_SupportedFormats();
  for (int format = Image_Format_UNKNOWN + 1; format < Image_Format_NB; ++format)
    formats->Add(static_cast<Image_Format>(format));
  for (int format = Image_Format_NB; format < Image_CompressedFormat_NB; ++format)
    formats->Add(static_cast<Image_CompressedFormat>(format));
  if (material->HasPbrMaterial())
  {
    const auto& pbr = material->PbrMaterial();
    ValidateViewerTexture(pbr.BaseColorTexture, formats, validated);
    ValidateViewerTexture(pbr.MetallicRoughnessTexture, formats, validated);
    ValidateViewerTexture(pbr.EmissiveTexture, formats, validated);
    ValidateViewerTexture(pbr.OcclusionTexture, formats, validated);
    ValidateViewerTexture(pbr.NormalTexture, formats, validated);
  }
  if (material->HasCommonMaterial())
    ValidateViewerTexture(material->CommonMaterial().DiffuseTexture, formats, validated);
}
}
