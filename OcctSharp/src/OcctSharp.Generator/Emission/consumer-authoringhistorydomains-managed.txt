using OcctSharp.Interop;

namespace OcctSharp;

#pragma warning disable CS1591
[Flags]
public enum AuthoringHistorySourceIssue
{
    None = 0, NoSourceIdentity = 1, InvalidArgument = 2, OutsideCopiedArgument = 4,
    KindMismatch = 8, MultipleArguments = 16, LegacyIndexDiffers = 32
}
[Flags]
public enum AuthoringHistoryTargetIssue
{ None = 0, NoShape = 1, NoFinalResult = 2, OutsideFinalResult = 4 }
public readonly record struct AuthoringHistoryDomainRow(int HistoryIndex, int SourceArgumentIndex,
    int LegacySourceTopologyIndex, ShapeKind? SourceKind, int CopiedSourceTopologyIndex,
    int SourceArgumentMemberships, AuthoringHistorySourceIssue SourceIssues,
    ShapeKind? TargetKind, int FinalTargetTopologyIndex, AuthoringHistoryTargetIssue TargetIssues,
    AuthoringHistoryKind Kind)
{
    /// <summary>Single argument membership proved by native identity; not complete occurrence paths.</summary>
    public bool HasUniqueNativeCopySourceMembership => CopiedSourceTopologyIndex >= 0 &&
        SourceArgumentMemberships == 1 && (SourceIssues & ~AuthoringHistorySourceIssue.LegacyIndexDiffers) == AuthoringHistorySourceIssue.None;
    public bool IsNativeMemberOfFinalResult => FinalTargetTopologyIndex >= 0 && TargetIssues == AuthoringHistoryTargetIssue.None;
}

/// <summary>Owned values describing the actual History source identity in the captured copied
/// input domains and the target's native membership in the actual final result. Source and final
/// addresses belong to separate domains. This does not prove occurrence paths, serialization,
/// material composition or cross-author reference continuation.</summary>
public sealed class AuthoringHistoryDomainReport
{
    internal AuthoringHistoryDomainReport(Guid inputDomainId, int finalTopologyCount, AuthoringHistoryDomainRow[] rows)
    {
        InputDomainId = inputDomainId; FinalDomainId = Guid.NewGuid(); FinalTopologyCount = finalTopologyCount;
        Rows = Array.AsReadOnly(rows);
    }
    public int SchemaVersion { get; } = 1;
    public Guid InputDomainId { get; }
    public Guid FinalDomainId { get; }
    public int FinalTopologyCount { get; }
    public IReadOnlyList<AuthoringHistoryDomainRow> Rows { get; }
    public bool ProvesCompleteOccurrencePaths { get; }
    public bool ProvesComposedReferenceIdentity { get; }
}

internal static class AuthoringHistoryDomainReader
{
    internal static AuthoringHistoryDomainReport? Read(FeatureResultHandle result, AuthoringInputCopyReport? copy,
        IReadOnlyList<AuthoringHistoryItem> history)
    {
        NativeError.ThrowIfFailed(NativeMethods.AuthoringHistoryDomainInfo(result, out var info), "authoring_history_domain_info");
        if (info.SchemaVersion == 0 && info.RowCount == 0 && info.FinalTopologyCount == 0 && info.SourceArgumentCount == 0) return null;
        if (info.SchemaVersion != 1 || copy is null || info.RowCount != history.Count || info.RowCount is < 0 or > 1000000 ||
            info.FinalTopologyCount is < 0 or > 1000000 || info.SourceArgumentCount != copy.Arguments.Count)
            throw new InvalidOperationException("Native History-domain report has an invalid version, count or copy domain.");
        AuthoringHistoryDomainRow[] rows = new AuthoringHistoryDomainRow[info.RowCount];
        for (int i = 0; i < rows.Length; i++)
        {
            NativeError.ThrowIfFailed(NativeMethods.AuthoringHistoryDomainRow(result, i, out var row), "authoring_history_domain_row");
            var item = history[i]; var source = item.Source;
            if (row.HistoryIndex != i || row.Reserved != 0 || row.Kind != (int)item.Kind ||
                row.SourceArgument < -1 || row.SourceArgument >= info.SourceArgumentCount ||
                row.LegacySourceIndex < -1 || row.CopyTargetIndex < -1 || row.SourceKind < -1 ||
                row.SourceMemberships is < 0 || row.SourceMemberships > info.SourceArgumentCount ||
                (row.SourceIssues & ~63) != 0 || (row.TargetIssues & ~7) != 0 ||
                row.TargetKind < -1 || row.TargetFinalIndex < -1 || row.TargetFinalIndex >= info.FinalTopologyCount ||
                (row.SourceKind >= 0 && !Enum.IsDefined((ShapeKind)row.SourceKind)) ||
                (row.TargetKind >= 0 && !Enum.IsDefined((ShapeKind)row.TargetKind)))
                throw new InvalidOperationException("Native History-domain row has invalid addresses or flags.");
            if (source is { } actual && (row.SourceArgument != actual.ArgumentIndex || row.LegacySourceIndex != actual.TopologyIndex || row.SourceKind != (int)actual.Kind))
                throw new InvalidOperationException("Native History-domain row differs from the actual legacy History source.");
            var sourceIssues = (AuthoringHistorySourceIssue)row.SourceIssues;
            var targetIssues = (AuthoringHistoryTargetIssue)row.TargetIssues;
            bool noSource = sourceIssues.HasFlag(AuthoringHistorySourceIssue.NoSourceIdentity);
            bool invalidArgument = sourceIssues.HasFlag(AuthoringHistorySourceIssue.InvalidArgument);
            bool outsideSource = sourceIssues.HasFlag(AuthoringHistorySourceIssue.OutsideCopiedArgument);
            if (noSource && (row.CopyTargetIndex != -1 || row.SourceMemberships != 0))
                throw new InvalidOperationException("Missing native source identity has an invented copy address.");
            if (!noSource && invalidArgument != (row.SourceArgument < 0 || row.SourceArgument >= info.SourceArgumentCount))
                throw new InvalidOperationException("Native source argument diagnostics contradict the actual address.");
            if (!noSource && !invalidArgument && (outsideSource != (row.CopyTargetIndex < 0) ||
                row.CopyTargetIndex >= copy.Arguments[row.SourceArgument].TargetTopologyCount))
                throw new InvalidOperationException("Native source membership is outside its captured copy domain.");
            if (sourceIssues.HasFlag(AuthoringHistorySourceIssue.MultipleArguments) != (row.SourceMemberships > 1) ||
                (!noSource && !invalidArgument && !outsideSource && row.SourceMemberships == 0) ||
                (sourceIssues.HasFlag(AuthoringHistorySourceIssue.LegacyIndexDiffers) && (row.CopyTargetIndex < 0 || row.CopyTargetIndex == row.LegacySourceIndex)))
                throw new InvalidOperationException("Native source membership diagnostics are inconsistent.");
            bool noShape = targetIssues.HasFlag(AuthoringHistoryTargetIssue.NoShape);
            bool noFinal = targetIssues.HasFlag(AuthoringHistoryTargetIssue.NoFinalResult);
            bool outsideFinal = targetIssues.HasFlag(AuthoringHistoryTargetIssue.OutsideFinalResult);
            if (noShape != (item.Shape is null) || noShape != (row.TargetKind == -1) || noFinal != (info.FinalTopologyCount == 0) ||
                (row.TargetFinalIndex == -1) != (noShape || noFinal || outsideFinal) ||
                (outsideFinal && (noShape || noFinal)))
                throw new InvalidOperationException("Native final-result membership contradicts the actual History target.");
            rows[i] = new(i, row.SourceArgument, row.LegacySourceIndex,
                row.SourceKind >= 0 ? (ShapeKind)row.SourceKind : null, row.CopyTargetIndex, row.SourceMemberships, sourceIssues,
                row.TargetKind >= 0 ? (ShapeKind)row.TargetKind : null, row.TargetFinalIndex, targetIssues, (AuthoringHistoryKind)row.Kind);
        }
        return new(copy.TargetDomainId, info.FinalTopologyCount, rows);
    }
}
#pragma warning restore CS1591
