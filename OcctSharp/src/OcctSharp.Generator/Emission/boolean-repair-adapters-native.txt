inline void RequireBooleanRepairHistorySource(const TopoDS_Shape& shape) {
 if(shape.IsNull()||(shape.ShapeType()!=TopAbs_FACE&&shape.ShapeType()!=TopAbs_EDGE&&shape.ShapeType()!=TopAbs_VERTEX&&shape.ShapeType()!=TopAbs_SOLID))
  throw std::invalid_argument("Boolean/repair history supports only nonnull faces, edges, vertices and solids.");
}
// Containers own their storage. Shapes retain exact private-graph identities;
// this is not a geometry deep copy or a cross-file persistent-name mapping.
inline std::vector<TopoDS_Shape> CopyBooleanRepairList(const NCollection_List<TopoDS_Shape>& source) {
 if(source.Size()>1000000)throw std::length_error("Boolean/repair history list exceeds its bounded limit.");
 std::vector<TopoDS_Shape> result;result.reserve(static_cast<size_t>(source.Size()));
 for(const auto& value:source)result.push_back(value);
 return result;
}
class BooleanDiagnosticBuffer final : public std::streambuf {
 std::string content;
 bool exhausted=false;
protected:
 int_type overflow(int_type value) override {
  if(traits_type::eq_int_type(value,traits_type::eof()))return traits_type::not_eof(value);
  if(content.size()==Limit){exhausted=true;return traits_type::eof();}
  content.push_back(traits_type::to_char_type(value));return value;
 }
 std::streamsize xsputn(const char* values,std::streamsize count) override {
  if(count<0||static_cast<unsigned long long>(count)>Limit-content.size()){exhausted=true;return 0;}
  if(count)content.append(values,static_cast<size_t>(count));return count;
 }
public:
 static constexpr size_t Limit=65536;
 std::string Copy(bool streamGood=true) const {
  if(exhausted||!streamGood)throw std::length_error("Boolean diagnostic text exceeds its bounded limit.");
  return content;
 }
};
struct BooleanDiagnosticCopies {
 bool hasErrors=false,hasWarnings=false;
 std::string errors,warnings;
};
// Owning call-private builder. No SDK reference, mutable receiver or stream
// crosses the ABI; callers keep their existing input graph and publication.
class BooleanHistoryBuilder final {
 std::unique_ptr<BRepAlgoAPI_BooleanOperation> value;
 template<class F> static std::string CopyDiagnostic(F&& write) {
  BooleanDiagnosticBuffer buffer;std::ostream stream(&buffer);
  stream.exceptions(std::ios::badbit|std::ios::failbit);
  write(stream);return buffer.Copy(stream.good());
 }
public:
 explicit BooleanHistoryBuilder(int operation) {
  if(operation==0)value=std::make_unique<BRepAlgoAPI_Fuse>();
  else if(operation==1)value=std::make_unique<BRepAlgoAPI_Cut>();
  else if(operation==2)value=std::make_unique<BRepAlgoAPI_Common>();
  else throw std::invalid_argument("Unsupported Boolean history operation.");
 }
 void SetArguments(const NCollection_List<TopoDS_Shape>& shapes){value->SetArguments(shapes);}
 void SetTools(const NCollection_List<TopoDS_Shape>& shapes){value->SetTools(shapes);}
 void SetNonDestructive(bool enabled){value->SetNonDestructive(enabled);}
 void SetRunParallel(bool enabled){value->SetRunParallel(enabled);}
 void SetToFillHistory(bool enabled){value->SetToFillHistory(enabled);}
 void Build(){value->Build();}
 bool IsDone() const {return value->IsDone();}
 bool HasErrors() const {return value->HasErrors();}
 bool HasWarnings() const {return value->HasWarnings();}
 TopoDS_Shape Shape(){return value->Shape();}
 std::vector<TopoDS_Shape> Modified(const TopoDS_Shape& shape){RequireBooleanRepairHistorySource(shape);return CopyBooleanRepairList(value->Modified(shape));}
 std::vector<TopoDS_Shape> Generated(const TopoDS_Shape& shape){RequireBooleanRepairHistorySource(shape);return CopyBooleanRepairList(value->Generated(shape));}
 bool IsDeleted(const TopoDS_Shape& shape){RequireBooleanRepairHistorySource(shape);return value->IsDeleted(shape);}
 BooleanDiagnosticCopies Diagnostics() const {
  BooleanDiagnosticCopies result;result.hasErrors=HasErrors();result.hasWarnings=HasWarnings();
  result.errors=CopyDiagnostic([&](std::ostream& stream){value->DumpErrors(stream);});
  result.warnings=CopyDiagnostic([&](std::ostream& stream){value->DumpWarnings(stream);});
  return result;
 }
};
inline std::vector<TopoDS_Shape> RepairHistoryModified(const BRepTools_History& history,const TopoDS_Shape& shape){RequireBooleanRepairHistorySource(shape);return CopyBooleanRepairList(history.Modified(shape));}
inline std::vector<TopoDS_Shape> RepairHistoryGenerated(const BRepTools_History& history,const TopoDS_Shape& shape){RequireBooleanRepairHistorySource(shape);return CopyBooleanRepairList(history.Generated(shape));}
inline bool RepairHistoryRemoved(const BRepTools_History& history,const TopoDS_Shape& shape){RequireBooleanRepairHistorySource(shape);return history.IsRemoved(shape);}
}
