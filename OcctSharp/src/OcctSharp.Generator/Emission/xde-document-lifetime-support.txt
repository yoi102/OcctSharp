#pragma once
#include <TDocStd_Document.hxx>
namespace OcctSharpXdeWorkspaces {
// SDK member destruction releases Data before undo/redo lists. Naming deltas must
// be cleared while labels/Data are alive, through the SDK BeforeClose protocol.
class CloseSafeDocument final:public TDocStd_Document {
public:
 CloseSafeDocument():TDocStd_Document(TCollection_ExtendedString("BinXCAF")){}
 ~CloseSafeDocument() override {BeforeClose();}
};
inline bool HasCloseProtocol(const occ::handle<TDocStd_Document>& document,int domain){return domain==1||dynamic_cast<CloseSafeDocument*>(document.get())!=nullptr;}
}
