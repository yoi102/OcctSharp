OcctSharp_Status OCCTSHARP_CALL occtsharp_repair_replace(const void* token,const void* inputs,const void* outputs,int32_t operation) {
 return Guard([&]{Require(operation>=0 && operation<=2,"Invalid replacement operation.");auto in=OcctSharpOwned::Lease<const ShapeSet>(inputs);auto out=OcctSharpOwned::Lease<const ShapeSet>(outputs);
  Require(!in->values.empty() && in->values.size()<=256 && out->values.size()<=256,"Invalid replacement counts.");
  Require(operation==1?out->values.empty():!out->values.empty(),"Replacement output mismatch.");Require(operation==2?out->values.size()==1:in->values.size()==1,"Replacement input mismatch.");
  auto session=OcctSharpOwned::Lease<Session>(token);std::lock_guard<std::mutex> lock(session->mutex);session->Live();Require(session->kind==5,"Explicit replacements require a replacement session.");
  ShapeKeys unique;NCollection_List<TopoDS_Shape> parts;auto type=in->values[0].ShapeType();
  for(const auto& s:in->values){Require(s.ShapeType()==type && unique.Add(s),"Replacement inputs must be distinct and have one type.");parts.Append(session->Exact(s));}
  for(const auto& s:out->values)Require(s.ShapeType()==type,"Replacement outputs must match input type.");
  Require(out->values.size()<=1 || type!=TopAbs_VERTEX,"Native history cannot retain a multi-vertex replacement carrier.");
  TopoDS_Shape replacement;BRep_Builder builder;
  if(!out->values.empty()){TopoDS_Compound all;builder.MakeCompound(all);for(const auto& s:out->values)builder.Add(all,s);BRepBuilderAPI_Copy copy(all,true,false);Require(copy.IsDone(),"Replacement copy failed.");
   if(out->values.size()==1)replacement=copy.ModifiedShape(out->values[0]);
   else{if(type==TopAbs_EDGE){TopoDS_Wire carrier;builder.MakeWire(carrier);replacement=carrier;}
    else if(type==TopAbs_FACE){TopoDS_Shell carrier;builder.MakeShell(carrier);replacement=carrier;}
    else if(type==TopAbs_SOLID){TopoDS_CompSolid carrier;builder.MakeCompSolid(carrier);replacement=carrier;}
    else{TopoDS_Compound carrier;builder.MakeCompound(carrier);replacement=carrier;}
    for(const auto& s:out->values)builder.Add(replacement,copy.ModifiedShape(s));}}
  Mutate(*session,[&]{if(operation==1)session->context->Remove(parts.First());else if(operation==0)session->context->Replace(parts.First(),replacement);else session->context->Merge(parts,replacement);session->changed=true;});});
}

OcctSharp_Status OCCTSHARP_CALL occtsharp_repair_session_snapshot(const void* token,void** result) {
 if(!result)return OCCTSHARP_STATUS_INVALID_ARGUMENT;*result=nullptr;
 return Guard([&]{auto session=OcctSharpOwned::Lease<Session>(token);std::lock_guard<std::mutex> lock(session->mutex);session->Live();
  try{auto current=session->Current();Require(!current.IsNull(),"Null repaired shape cannot be published.");auto nativeHistory=session->context->History();auto history=std::make_shared<History>();
   BRep_Builder builder;TopoDS_Compound all;builder.MakeCompound(all);builder.Add(all,current);int64_t count=1;
   for(int i=1;i<=session->originals.Extent();++i){HistoryRow row;row.original=session->originals(i);auto privateShape=session->Exact(row.original);TopoDS_Shape direct;
    row.status=session->context->Status(privateShape,direct,false);row.supported=BRepTools_History::IsSupportedType(privateShape);
    auto leaf=session->context->ValueLeaf(privateShape);if(!leaf.IsNull()){row.leaf.push_back(leaf);row.newShape=session->context->IsNewShape(leaf);builder.Add(all,leaf);Require(++count<=MaxItems*8,"History too large.");}
    if(row.supported){row.removed=nativeHistory->IsRemoved(privateShape);
     for(NCollection_List<TopoDS_Shape>::Iterator j(nativeHistory->Modified(privateShape));j.More();j.Next()){row.modified.push_back(j.Value());builder.Add(all,j.Value());Require(++count<=MaxItems*8,"History too large.");}
     for(NCollection_List<TopoDS_Shape>::Iterator j(nativeHistory->Generated(privateShape));j.More();j.Next()){row.generated.push_back(j.Value());builder.Add(all,j.Value());Require(++count<=MaxItems*8,"History too large.");}}
    row.evidence=!row.supported?3:row.removed?1:!row.modified.empty() || !row.generated.empty()?2:session->changed?3:0;history->rows.push_back(std::move(row));}
   BRepBuilderAPI_Copy freeze(all,true,false);Require(freeze.IsDone(),"Final repair snapshot failed.");history->result=freeze.ModifiedShape(current);history->translated=new BRepTools_History();
   for(auto& row:history->rows){for(auto* list:{&row.modified,&row.generated,&row.leaf})for(auto& s:*list)s=freeze.ModifiedShape(s);
    if(row.removed)history->translated->Remove(row.original);

   Publication<History> publication(history);*result=publication.Commit();}
  catch(...){session->retired=true;throw;}});
}

OcctSharp_Status OCCTSHARP_CALL occtsharp_repair_edge_fix(const OcctSharp_ShapeHandle* input,const OcctSharp_ShapeHandle* support,int32_t operation,double tolerance,int32_t seam,void** result) {
 if(!result)return OCCTSHARP_STATUS_INVALID_ARGUMENT;*result=nullptr;
 return Guard([&]{CheckPrecision(tolerance);Flag(seam);Require(operation>=0 && operation<=8,"Unknown edge fix.");auto original=Typed(input,TopAbs_EDGE);TopoDS_Shape face;

  BRep_Builder builder;TopoDS_Compound all;builder.MakeCompound(all);builder.Add(all,original);

  BRepBuilderAPI_Copy copy(all,true,false);Require(copy.IsDone(),"Edge copy failed.");auto edge=TopoDS::Edge(copy.ModifiedShape(original));TopoDS_Face f;if(needsFace)f=TopoDS::Face(copy.ModifiedShape(face));ShapeFix_Edge tool;bool changed=false;
  switch(operation){case 0:changed=tool.FixAddCurve3d(edge);break;case 1:changed=tool.FixAddPCurve(edge,f,seam!=0,tolerance);break;

   case 4:changed=tool.FixSameParameter(edge,tolerance);break;case 5:changed=tool.FixSameParameter(edge,f,tolerance);break;

OcctSharp_Status OCCTSHARP_CALL occtsharp_repair_edge_split(const OcctSharp_ShapeHandle* input,const OcctSharp_ShapeHandle* support,const OcctSharp_ShapeHandle* vertex1,const OcctSharp_ShapeHandle* vertex2,int32_t operation,double first,double last,double tolerance3d,double tolerance2d,void** result) {

  TopoDS_Shape v1,v2;if(operation!=2)v1=Typed(vertex1,TopAbs_VERTEX);if(operation==3)v2=Typed(vertex2,TopAbs_VERTEX);
  BRep_Builder builder;TopoDS_Compound all;builder.MakeCompound(all);builder.Add(all,face);if(!v1.IsNull())builder.Add(all,v1);if(!v2.IsNull())builder.Add(all,v2);
