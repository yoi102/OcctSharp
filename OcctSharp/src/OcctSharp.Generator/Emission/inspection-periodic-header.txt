#pragma once
#include <BRepExtrema_DistShapeShape.hxx>
#include <TopoDS_Shape.hxx>
#include <gp_Pnt.hxx>
#include <vector>

// Call-local corrections retain ORIGINAL supports and ORIGINAL edge parameters.
// Completes vertex/periodic-edge projection and refines native interior periodic
// extrema on their actual original supports. Geometry, ranges and pcurves stay
// unchanged; native candidate discovery and containment semantics remain.
class OcctSharp_PeriodicExactDistance final
{
  struct Witness
  {
    gp_Pnt firstPoint,secondPoint;
    TopoDS_Shape firstSupport,secondSupport;
    double edgeParameter;
    bool edgeOnFirst;
  };
  struct InteriorWitness
  {
    gp_Pnt firstPoint,secondPoint;
    TopoDS_Shape firstSupport,secondSupport;
    BRepExtrema_SupportType firstType,secondType;
    double firstU=0,firstV=0,secondU=0,secondV=0;
  };
  std::vector<InteriorWitness> myInteriors;
  void RefinePeriodicInteriors(const TopoDS_Shape& first,const TopoDS_Shape& second);
  const InteriorWitness& Interior(int index) const;
  BRepExtrema_DistShapeShape myNative;
  std::vector<Witness> myCorrections;
  double myDistance;
  void CompleteVertexEdge(const TopoDS_Shape& vertices,const TopoDS_Shape& edges,bool edgeOnFirst);
  void Accept(Witness witness,double distance);
  const Witness& Correction(int index) const;
public:
  OcctSharp_PeriodicExactDistance(const TopoDS_Shape& first,const TopoDS_Shape& second);
  bool IsDone() const;
  int NbSolution() const;
  double Value() const;
  bool InnerSolution() const;
  gp_Pnt PointOnShape1(int index) const;
  gp_Pnt PointOnShape2(int index) const;
  TopoDS_Shape SupportOnShape1(int index) const;
  TopoDS_Shape SupportOnShape2(int index) const;
  BRepExtrema_SupportType SupportTypeShape1(int index) const;
  BRepExtrema_SupportType SupportTypeShape2(int index) const;
  void ParOnEdgeS1(int index,double& parameter) const;
  void ParOnEdgeS2(int index,double& parameter) const;
  void ParOnFaceS1(int index,double& u,double& v) const;
  void ParOnFaceS2(int index,double& u,double& v) const;
};
