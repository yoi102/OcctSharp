/* A copied prefix chain. Kind1 is an unchanged legacy primitive;kind2 is an
   exact signed-right-normal offset of an earlier node;kind3 is exact native
   parameter continuation of an earlier node (signed_offset=0). No retained pointer is
   stored. Legacy OcctSharp_SketchCurve layout andentry points stay unchanged. */
typedef struct OcctSharp_SketchExpressionNode {
  int32_t kind;
  int32_t basis_index;
  double signed_offset;
  OcctSharp_SketchCurve curve;
} OcctSharp_SketchExpressionNode;

OCCTSHARP_API OcctSharp_Status OCCTSHARP_CALL occtsharp_sketch_expression_evaluate(
  const OcctSharp_SketchExpressionNode*,int32_t,double,OcctSharp_SketchEvaluation*);
OCCTSHARP_API OcctSharp_Status OCCTSHARP_CALL occtsharp_sketch_expression_derivative(
  const OcctSharp_SketchExpressionNode*,int32_t,double,int32_t,OcctSharp_SketchPoint2d*);
OCCTSHARP_API OcctSharp_Status OCCTSHARP_CALL occtsharp_sketch_expression_make_edge(
  const OcctSharp_SketchExpressionNode*,int32_t,const OcctSharp_SketchPlane*,OcctSharp_ShapeHandle**);
/* Query with results=null/capacity=0, then fill caller-owned buffers. Outputs
   cannot overlap either input chain, its primitive tables, or other outputs.
   All outputs remain unchanged on failure. A null second chain with count=0
   requests self intersection; two chains always classify the two operands. */
OCCTSHARP_API OcctSharp_Status OCCTSHARP_CALL occtsharp_sketch_expression_project(
  const OcctSharp_SketchExpressionNode*,int32_t,OcctSharp_SketchPoint2d,
  OcctSharp_SketchProjection*,int32_t,int32_t*);
OCCTSHARP_API OcctSharp_Status OCCTSHARP_CALL occtsharp_sketch_expression_intersect(
  const OcctSharp_SketchExpressionNode*,int32_t,const OcctSharp_SketchExpressionNode*,int32_t,
  double,OcctSharp_SketchIntersection*,int32_t,int32_t*,int32_t*,int32_t*);
OCCTSHARP_API OcctSharp_Status OCCTSHARP_CALL occtsharp_sketch_expression_profile_face(
  const OcctSharp_ShapeHandle*,const OcctSharp_ShapeHandle* const*,int32_t,
  const OcctSharp_SketchPlane*,double,OcctSharp_ShapeHandle**);
OCCTSHARP_API OcctSharp_Status OCCTSHARP_CALL occtsharp_sketch_expression_wire_contains(
  const OcctSharp_ShapeHandle*,OcctSharp_SketchPoint2d,double,int32_t*);

/* Owned readback of the actual native tree, never an author provenance claim.
   Kinds: line1,circle2,ellipse3,Bezier4,BSpline5,trim6,offset7.
   Tables are flat caller-owned values. Node indices form a single prefix chain.
   Flags: rational1,periodic2. Conic X/Y axes retain native parameter handedness.
   Line origin/direction retain native line parameters; they are not endpoints. */
typedef struct OcctSharp_SketchNativeNode {
  int32_t kind,basis_index,degree,flags;
  int32_t pole_start,pole_count,knot_start,knot_count;
  double first_parameter,last_parameter,signed_offset,major_radius,minor_radius;
  OcctSharp_SketchPoint2d origin,x_direction,y_direction;
} OcctSharp_SketchNativeNode;
typedef struct OcctSharp_SketchNativeSummary {
  int32_t node_count,pole_count,knot_count,reversed;
  double first_parameter,last_parameter;
} OcctSharp_SketchNativeSummary;
/* Query with every buffer null/capacity0; fill all buffers together. All outputs
   are mutually disjoint and disjoint from shape/plane. Any failure writes none. */
OCCTSHARP_API OcctSharp_Status OCCTSHARP_CALL occtsharp_sketch_expression_read_edge(
  const OcctSharp_ShapeHandle*,const OcctSharp_SketchPlane*,double,
  OcctSharp_SketchNativeSummary*,OcctSharp_SketchNativeNode*,int32_t,
  OcctSharp_SketchPoint2d*,double*,int32_t,double*,int32_t*,int32_t);
/* Reconstruct solely from the copied snapshot, with no retained native object.
   Parameter is in the directed finite edge domain. Orders1..3 use exact analytic derivatives. */
OCCTSHARP_API OcctSharp_Status OCCTSHARP_CALL occtsharp_sketch_expression_snapshot_evaluate(
  const OcctSharp_SketchNativeSummary*,const OcctSharp_SketchNativeNode*,
  const OcctSharp_SketchPoint2d*,const double*,const double*,const int32_t*,
  double,int32_t,OcctSharp_SketchEvaluation*);
