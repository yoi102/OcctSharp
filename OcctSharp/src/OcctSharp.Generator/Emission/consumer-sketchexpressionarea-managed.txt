namespace OcctSharp;

internal static class SketchExpressionArea
{
    internal static double Integrate(SketchCurveExpression2d curve, SketchPoint2d origin)
    {
        var leaf = curve;
        while (leaf.Basis is not null) leaf = leaf.Basis;
        double span = curve.LastParameter - curve.FirstParameter;
        var breaks = leaf.Primitive!.Knots.Select(knot => (knot - curve.FirstParameter) / span)
            .Select(value => curve.Reversed ? 1 - value : value)
            .Where(value => value > 0 && value < 1).Append(0).Append(1).Order().Distinct().ToArray();
        int evaluations = 0;
        double total = 0, compensation = 0;
        for (int i = 0; i + 1 < breaks.Length; ++i)
        {
            double a = breaks[i], b = breaks[i + 1];
            double whole = Gauss(a, b, true);
            double value = Adaptive(a, b, 1e-10 * Math.Max(1, Math.Abs(whole)), 20) - compensation;
            double next = total + value;
            compensation = (next - total) - value;
            total = next;
        }
        return total;

        double Value(double t)
        {
            if (++evaluations > 2000000)
                throw new InvalidOperationException("Exact expression area integration exceeded its evaluation budget.");
            var sample = curve.Evaluate(t);
            double value = .5 * ((sample.Point.X - origin.X) * sample.FirstDerivative.Y -
                (sample.Point.Y - origin.Y) * sample.FirstDerivative.X) * span;
            if (!double.IsFinite(value)) throw new InvalidOperationException("Exact expression area integrand is not finite.");
            return value;
        }

        double Gauss(double a, double b, bool high)
        {
            // Interior nodes avoid knot derivatives. The reference origin is
            // shared by the whole loop to avoid world-coordinate cancellation.
            ReadOnlySpan<double> nodes = high
                ? [.9894009349916499,.9445750230732326,.8656312023878318,.7554044083550030,
                   .6178762444026438,.4580167776572274,.2816035507792589,.0950125098376374]
                : [.9602898564975363,.7966664774136267,.5255324099163290,.1834346424956498];
            ReadOnlySpan<double> weights = high
                ? [.02715245941175409,.06225352393864789,.09515851168249278,.12462897125553387,
                   .14959598881657673,.16915651939500254,.18260341504492359,.18945061045506850]
                : [.1012285362903763,.2223810344533745,.3137066458778873,.3626837833783620];
            double middle = (a + b) / 2, half = (b - a) / 2, sum = 0;
            for (int i = 0; i < nodes.Length; ++i)
                sum += weights[i] * (Value(middle - half * nodes[i]) + Value(middle + half * nodes[i]));
            return half * sum;
        }

        double Adaptive(double a, double b, double tolerance, int depth)
        {
            double lower = Gauss(a, b, false), higher = Gauss(a, b, true);
            if (Math.Abs(higher - lower) <= tolerance) return higher;
            if (depth == 0) throw new InvalidOperationException("Exact expression area integration did not converge.");
            double middle = (a + b) / 2;
            if (middle == a || middle == b)
                throw new InvalidOperationException("Exact expression area integration exhausted parameter precision.");
            return Adaptive(a, middle, tolerance / 2, depth - 1) + Adaptive(middle, b, tolerance / 2, depth - 1);
        }
    }
}
