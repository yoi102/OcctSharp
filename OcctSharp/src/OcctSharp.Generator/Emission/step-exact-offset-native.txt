  bool SameUnit(const double first,const double second)
  {
    return std::isfinite(first)&&std::isfinite(second)&&first>0&&second>0
      &&std::abs(first-second)<=64*std::numeric_limits<double>::epsilon()*std::max(first,second);
  }
  StepData_Factors ContextFactors(const occ::handle<StepRepr_Representation>& representation,
    const double cascadeUnit)
  {
    if(representation.IsNull()||!std::isfinite(cascadeUnit)||cascadeUnit<=0)
      throw std::runtime_error("Exact STEP units need a representation and finite positive system length unit.");
    StepData_Factors local,result;local.SetCascadeUnit(cascadeUnit);
      const auto context=representation->ContextOfItems();
      auto units=occ::down_cast<StepRepr_GlobalUnitAssignedContext>(context);
      const auto two=occ::down_cast<StepGeom_GeometricRepresentationContextAndGlobalUnitAssignedContext>(context);
      const auto three=occ::down_cast<StepGeom_GeomRepContextAndGlobUnitAssCtxAndGlobUncertaintyAssCtx>(context);
      if(!two.IsNull())units=two->GlobalUnitAssignedContext();
      if(!three.IsNull())units=three->GlobalUnitAssignedContext();
      if(units.IsNull())throw std::runtime_error("Exact STEP units have no assigned representation context.");
      if(units->Units().IsNull()||units->NbUnits()==0)
        throw std::runtime_error("Exact STEP units have an empty assigned representation context.");
      STEPConstruct_UnitContext decoded;
      for(int index=1;index<=units->NbUnits();++index)
      {
        // OCCT diagnoses repeated length assignments, but silently overwrites
        // repeated angle assignments. Decode each real unit independently to
        // prove its category before accumulating the representation's factors.
        STEPConstruct_UnitContext item;
        if(item.ComputeFactors(units->UnitsValue(index),local)!=0
          ||(item.LengthDone()&&decoded.LengthDone())
          ||(item.PlaneAngleDone()&&decoded.PlaneAngleDone())
          ||(item.SolidAngleDone()&&decoded.SolidAngleDone())
          ||decoded.ComputeFactors(units->UnitsValue(index),local)!=0)
          throw std::runtime_error("Exact STEP units contain an invalid or repeated unit assignment.");
      }
      if((decoded.PlaneAngleDone()&&!SameUnit(decoded.PlaneAngleFactor(),decoded.PlaneAngleFactor()))
        ||(decoded.SolidAngleDone()&&!SameUnit(decoded.SolidAngleFactor(),decoded.SolidAngleFactor())))
        throw std::runtime_error("Exact STEP units produced an invalid angle factor.");
      if(!decoded.LengthDone())throw std::runtime_error("Exact STEP units lack a proved length assignment.");
    result.InitializeFactors(decoded.LengthFactor(),decoded.PlaneAngleFactor(),decoded.SolidAngleFactor());
    if(!SameUnit(result.LengthFactor(),result.LengthFactor()))
      throw std::runtime_error("Exact STEP units produced an invalid length factor.");
    return result;
  }
  bool SameFactors(const StepData_Factors& first,const StepData_Factors& second)
  {
    return SameUnit(first.LengthFactor(),second.LengthFactor())
      &&SameUnit(first.PlaneAngleFactor(),second.PlaneAngleFactor())
      &&SameUnit(first.SolidAngleFactor(),second.SolidAngleFactor());
  }
  using ContextEntry=std::pair<occ::handle<StepRepr_Representation>,StepData_Factors>;
  std::vector<ContextEntry> RepresentationContexts(const Interface_Graph& graph,
    const occ::handle<Standard_Transient>& entity,const double cascadeUnit)
  {
    std::vector<ContextEntry> result;
    // EACH branch stops at its first representation; assembly units must not
    // replace the geometry representation's own assigned units.
    for(auto representations=graph.TypedSharings(entity,STANDARD_TYPE(StepRepr_Representation));
      representations.More();representations.Next())
    {
      const auto representation=occ::down_cast<StepRepr_Representation>(representations.Value());
      result.emplace_back(representation,ContextFactors(representation,cascadeUnit));
    }
    return result;
  }
  StepData_Factors RepresentationFactors(const Interface_Graph& graph,
    const occ::handle<Standard_Transient>& entity,const double cascadeUnit)
  {
    const auto contexts=RepresentationContexts(graph,entity,cascadeUnit);
    if(contexts.empty())throw std::runtime_error("Exact STEP units have no reachable geometry representation.");
    for(const auto& entry:contexts)
      if(!SameFactors(contexts.front().second,entry.second))
        throw std::runtime_error("Exact STEP units are ambiguous across shared representation contexts.");
    return contexts.front().second;
  }
  StepData_Factors WriterFactors(const Interface_Graph& graph,
    const occ::handle<Standard_Transient>& entity,const occ::handle<StepData_StepModel>& model)
  {
    const auto factors=RepresentationFactors(graph,entity,model->LocalLengthUnit());
    // Prove agreement with the transfer that already created the basis and
    // endpoints. Changing metadata AFTER transfer cannot rescale that geometry.
    if(!SameUnit(factors.LengthFactor(),model->WriteLengthUnit()/model->LocalLengthUnit()))
      throw std::runtime_error("Exact STEP units disagree with the completed writer transfer.");
    return factors;
  }
  bool HasOffset(const occ::handle<Geom_Curve>& curve, const int depth = 0)
  {
    if (curve.IsNull() || depth > 32) throw std::runtime_error("Invalid or over-depth exact STEP curve.");
    if (!occ::down_cast<Geom_OffsetCurve>(curve).IsNull()) return true;
    const auto trimmed = occ::down_cast<Geom_TrimmedCurve>(curve);
    return !trimmed.IsNull() && HasOffset(trimmed->BasisCurve(), depth + 1);
  }

  occ::handle<StepGeom_Curve> ExactCurve(const occ::handle<Geom_Curve>& curve,
    const StepData_Factors& factors, const int depth = 0)
  {
    if (curve.IsNull() || depth > 32) throw std::runtime_error("Invalid or over-depth exact STEP basis.");
    const auto offset = occ::down_cast<Geom_OffsetCurve>(curve);
    if (!offset.IsNull())
    {
      if (!std::isfinite(offset->Offset())) throw std::runtime_error("Non-finite exact STEP offset.");
      const auto basis = ExactCurve(offset->BasisCurve(), factors, depth + 1);
      GeomToStep_MakeDirection direction(offset->Direction());
      occ::handle<StepGeom_OffsetCurve3d> result = new StepGeom_OffsetCurve3d;
      // OCCT's STEP reader applies a minus sign to OFFSET_CURVE_3D distance.
      // Unknown self-intersection is retained; this converter invents no proof.
      const double distance=-offset->Offset()/factors.LengthFactor();
      if(!std::isfinite(distance)||(offset->Offset()!=0&&distance==0))
        throw std::runtime_error("Exact STEP units overflowed or underflowed a written offset distance.");
      result->Init(new TCollection_HAsciiString(""), basis,
        distance, StepData_LUnknown, direction.Value());
      return result;
    }
    const auto trimmed = occ::down_cast<Geom_TrimmedCurve>(curve);
    if (!trimmed.IsNull() && HasOffset(trimmed->BasisCurve()))
      return ExactCurve(trimmed->BasisCurve(), factors, depth + 1);
    GeomToStep_MakeCurve maker(curve, factors);
    const auto result = maker.Value();
    if (result.IsNull()) throw std::runtime_error("Exact STEP basis conversion produced no curve.");
    return result;
  }

  template<class T> occ::handle<T> Bound(const occ::handle<Transfer_FinderProcess>& process,
    const TopoDS_Shape& source)
  {
    // The actual transfer binding is the sole correspondence proof. No index,
    // distance, normal or topology-count matching is permitted.
    const auto binder = process->Find(TransferBRep::ShapeMapper(process, source));
    const auto simple = occ::down_cast<Transfer_SimpleBinderOfTransient>(binder);
    if (simple.IsNull() || binder->IsMultiple() || !binder->NextResult().IsNull())
      throw std::runtime_error("Missing or ambiguous exact source-to-STEP transfer binding.");
    const auto result = occ::down_cast<T>(simple->Result());
    if (result.IsNull()) throw std::runtime_error("Exact source-to-STEP binding has the wrong result type.");
    return result;
  }
}

static OcctSharp_ExactStepOffsetReport CompleteExactStepOffsets(
  STEPControl_Writer& writer, const TopoDS_Shape& original,const BRepBuilderAPI_Copy* transferCopy,const bool reverseCopy=false)
{
  const auto model = writer.Model();
  const auto process = writer.WS()->TransferWriter()->FinderProcess();
  if (original.IsNull() || model.IsNull() || process.IsNull())
    throw std::runtime_error("Exact STEP completion requires an original shape and live transfer.");
  const Interface_Graph graph(model);
  OcctSharp_ExactStepOffsetReport report;
  NCollection_IndexedMap<TopoDS_Shape, TopTools_ShapeMapHasher> edges;
  TopExp::MapShapes(original, TopAbs_EDGE, edges);
  if (edges.Extent() > 100000) throw std::runtime_error("Exact STEP edge budget exceeded.");
  for (int index = 1; index <= edges.Extent(); ++index)
  {
    double first, last;
    const auto exactSource=reverseCopy?transferCopy->ModifiedShape(edges(index)):edges(index);
    if(exactSource.IsNull()||exactSource.ShapeType()!=TopAbs_EDGE)
      throw std::runtime_error("XDE transfer edge is outside the actual exact copy mapping.");
    const auto curve = BRep_Tool::Curve(TopoDS::Edge(exactSource), first, last);
    if (curve.IsNull() || !HasOffset(curve)) continue;
    const auto edge = Bound<StepShape_EdgeCurve>(process,transferCopy==nullptr||reverseCopy?edges(index):transferCopy->ModifiedShape(edges(index)));
    const auto factors=WriterFactors(graph,edge,model);
    const auto converted = ExactCurve(curve, factors);
    const auto surfaceCurve = occ::down_cast<StepGeom_SurfaceCurve>(edge->EdgeGeometry());
    if (surfaceCurve.IsNull()) edge->SetEdgeGeometry(converted);
    else surfaceCurve->SetCurve3d(converted);
    model->AddWithRefs(converted);
    ++report.RepairedEdges;
  }
  NCollection_IndexedMap<TopoDS_Shape, TopTools_ShapeMapHasher> faces;
  TopExp::MapShapes(original, TopAbs_FACE, faces);
  if (faces.Extent() > 100000) throw std::runtime_error("Exact STEP face budget exceeded.");
  for (int index = 1; index <= faces.Extent(); ++index)
  {
    const auto exactSource=reverseCopy?transferCopy->ModifiedShape(faces(index)):faces(index);
    if(exactSource.IsNull()||exactSource.ShapeType()!=TopAbs_FACE)
      throw std::runtime_error("XDE transfer face is outside the actual exact copy mapping.");
    auto surface = BRep_Tool::Surface(TopoDS::Face(exactSource));
    const auto trimmed = occ::down_cast<Geom_RectangularTrimmedSurface>(surface);
    if (!trimmed.IsNull()) surface = trimmed->BasisSurface();
    const auto extrusion = occ::down_cast<Geom_SurfaceOfLinearExtrusion>(surface);
    if (extrusion.IsNull() || !HasOffset(extrusion->BasisCurve())) continue;
    const auto face = Bound<StepShape_AdvancedFace>(process,transferCopy==nullptr||reverseCopy?faces(index):transferCopy->ModifiedShape(faces(index)));
    const auto factors=WriterFactors(graph,face,model);
    const auto converted = occ::down_cast<StepGeom_SurfaceOfLinearExtrusion>(face->FaceGeometry());
    if (converted.IsNull()) throw std::runtime_error("Exact offset extrusion transfer changed its surface kind.");
    const auto basis = ExactCurve(extrusion->BasisCurve(), factors);
    converted->SetSweptCurve(basis);
    model->AddWithRefs(basis);
    ++report.RepairedExtrusions;
  }
  for (int index = 1; index <= model->NbEntities(); ++index)
  {
    const auto swept = occ::down_cast<StepGeom_SweptSurface>(model->Value(index));
    if (!swept.IsNull() && swept->SweptCurve().IsNull())
      throw std::runtime_error("STEP graph contains an unresolved swept basis.");
    const auto offset = occ::down_cast<StepGeom_OffsetCurve3d>(model->Value(index));
    if (!offset.IsNull() && (offset->BasisCurve().IsNull() || offset->RefDirection().IsNull()
        || model->Number(offset->BasisCurve()) == 0 || model->Number(offset->RefDirection()) == 0))
      throw std::runtime_error("STEP graph contains an unresolved exact offset reference.");
  }
  return report;
}

OcctSharp_ExactStepOffsetReport OcctSharp_CompleteExactStepOffsets(
  STEPControl_Writer& writer,const TopoDS_Shape& original)
{
  return CompleteExactStepOffsets(writer,original,nullptr);
}

OcctSharp_ExactStepOffsetReport OcctSharp_CompleteTransferredExactStepOffsets(STEPControl_Writer& writer)
{
  const auto process=writer.WS()->TransferWriter()->FinderProcess();
  if(process.IsNull())throw std::runtime_error("Exact STEP completion requires actual transferred shape mappers.");
  TopoDS_Compound original;BRep_Builder builder;builder.MakeCompound(original);
  NCollection_IndexedMap<TopoDS_Shape,TopTools_ShapeMapHasher> sources;
  const int mapped=process->NbMapped();
  for(int index=1;index<=mapped;++index)
  {
    const auto mapper=occ::down_cast<TransferBRep_ShapeMapper>(process->Mapped(index));
    if(mapper.IsNull())continue;
    const auto& shape=mapper->Value();
    if(shape.IsNull()||(shape.ShapeType()!=TopAbs_EDGE&&shape.ShapeType()!=TopAbs_FACE))continue;
    if(sources.Contains(shape))continue;
    sources.Add(shape);builder.Add(original,shape);
  }
  return OcctSharp_CompleteExactStepOffsets(writer,original);
}

  double ReadOffsetDistance(const occ::handle<StepGeom_OffsetCurve3d>& offset,
    const StepData_Factors& factors)
  {
    const double value=offset->Distance()*factors.LengthFactor();
    if(!std::isfinite(offset->Distance())||!std::isfinite(value)
      ||(offset->Distance()!=0&&value==0))
      throw std::runtime_error("Exact STEP units overflowed or underflowed a read offset distance.");
    return value;
  }
  std::string ExactStepText(const occ::handle<StepData_StepModel>& model,
    const occ::handle<StepData_Protocol>& protocol)
  {
    StepData_StepWriter writer(model);writer.LabelMode()=0;
    // 17 significant decimal digits round-trip finite IEEE doubles. This is
    // entity serialization, never a geometric approximation or fitted curve.
    writer.FloatWriter().SetFormat("%.17g");writer.SendModel(protocol);
    if(!writer.CheckList().IsEmpty(true))throw std::runtime_error("Exact STEP private serialization has failed entities.");
    std::ostringstream output;output.imbue(std::locale::classic());
    if(!writer.Print(output))throw std::runtime_error("Exact STEP private serialization could not finish.");
    return output.str();
  }
  struct SerializedStepCopy
  {
    occ::handle<StepData_StepModel> Source,Model;
    std::unique_ptr<STEPControl_Reader> Reader;
    std::vector<occ::handle<Standard_Transient>> Identities;
    SerializedStepCopy(STEPControl_Reader& reader)
    {
      Source=reader.StepModel();
      const auto protocol=occ::down_cast<StepData_Protocol>(reader.WS()->Protocol());
      if(protocol.IsNull())throw std::runtime_error("Exact STEP serialization protocol is unavailable.");
      const auto text=ExactStepText(Source,protocol);
      std::istringstream input(text);input.imbue(std::locale::classic());Reader=std::make_unique<STEPControl_Reader>();
      if(Reader->ReadStream("OcctSharp private exact unit contexts",input)!=IFSelect_RetDone)
        throw std::runtime_error("Exact STEP private model could not be read.");
      Model=Reader->StepModel();
      if(Model.IsNull()||Model->NbEntities()!=Source->NbEntities()||ExactStepText(Model,protocol)!=text)
        throw std::runtime_error("Exact STEP private model changed its canonical entity content.");
      // These are actual Part21 identifiers assigned by LabelMode=0, not
      // inferred geometry or an assumed position in the imported entity list.
      Identities.resize(static_cast<size_t>(Source->NbEntities())+1);
      for(int index=1;index<=Model->NbEntities();++index)
      {
        const auto entity=Model->Value(index);const int label=Model->IdentLabel(entity);
        if(label<1||label>Source->NbEntities()||!Identities[label].IsNull()
          ||entity==Source->Value(label)||entity->DynamicType()!=Source->Value(label)->DynamicType())
          throw std::runtime_error("Exact STEP serialization lost a unique actual entity identity or type.");
        Identities[label]=entity;
      }
      Model->InternalParameters=Source->InternalParameters;
      Model->SetLocalLengthUnit(Source->LocalLengthUnit());Model->SetWriteLengthUnit(Source->WriteLengthUnit());
    }
    occ::handle<Standard_Transient> Counterpart(const occ::handle<Standard_Transient>& original) const
    {
      const int number=Source->Number(original);
      if(number<1||static_cast<size_t>(number)>=Identities.size()||Identities[number].IsNull())
        throw std::runtime_error("Exact STEP identity is outside the actual serialized entity mapping.");
      return Identities[number];
    }
  };
  struct UnitCopyGroup
  {
    StepData_Factors Factors;
    std::unique_ptr<SerializedStepCopy> Copy;

  std::unique_ptr<STEPControl_Reader> ContextLocalReader(STEPControl_Reader& reader,
    const std::vector<ContextEntry>& representations)
  {
    const auto source=reader.StepModel();const auto protocol=reader.WS()->Protocol();
    if(protocol.IsNull()||source->NbEntities()>100000||representations.size()>1000)
      throw std::runtime_error("Exact STEP context copy protocol or graph budget is unavailable.");
    SerializedStepCopy primary(reader);const auto target=primary.Model;
    std::vector<UnitCopyGroup> groups;
    std::unordered_set<const Standard_Transient*> affected,closure,geometry;
    for(const auto& entry:representations)affected.insert(entry.first.get());
    const Interface_Graph originalGraph(source);
    for(const auto& entry:representations)
    {
      auto group=std::find_if(groups.begin(),groups.end(),[&](const auto& value){return SameFactors(value.Factors,entry.second);});
      if(group==groups.end())
      {
        if((groups.size()+2)*static_cast<size_t>(source->NbEntities())>1000000)
          throw std::runtime_error("Exact STEP total context copy graph budget exceeded.");
        groups.push_back({entry.second,std::make_unique<SerializedStepCopy>(reader)});group=groups.end()-1;
      }
      const auto representation=occ::down_cast<StepRepr_Representation>(primary.Counterpart(entry.first));
      if(representation.IsNull()||entry.first->Items().IsNull())
        throw std::runtime_error("Invalid exact STEP representation item closure.");
      occ::handle<NCollection_HArray1<occ::handle<StepRepr_RepresentationItem>>> items=
        new NCollection_HArray1<occ::handle<StepRepr_RepresentationItem>>(1,entry.first->NbItems());
      std::vector<occ::handle<Standard_Transient>> pending;
      for(int index=1;index<=entry.first->NbItems();++index)
      {
        const auto original=entry.first->ItemsValue(index);
        const auto item=occ::down_cast<StepRepr_RepresentationItem>(group->Copy->Counterpart(original));
        if(item.IsNull())throw std::runtime_error("Exact STEP context copy lost an actual representation item identity.");
        items->SetValue(index,item);target->AddWithRefs(item,protocol);pending.push_back(original);
      }
      representation->SetItems(items);
      while(!pending.empty())
      {
        const auto original=pending.back();pending.pop_back();
        if(!closure.insert(original.get()).second)continue;
        if(source->Number(original)==0||affected.contains(original.get()))
          throw std::runtime_error("Exact STEP item closure crosses an unproved geometry representation identity.");
        if(original->IsKind(STANDARD_TYPE(StepGeom_GeometricRepresentationItem))
          ||original->IsKind(STANDARD_TYPE(StepShape_TopologicalRepresentationItem)))geometry.insert(original.get());
        for(auto children=originalGraph.Shareds(original);children.More();children.Next())pending.push_back(children.Value());
      }
    }
    for(const auto* original:geometry)
    {
      const occ::handle<Standard_Transient> entity(const_cast<Standard_Transient*>(original));
      for(auto parents=originalGraph.Sharings(entity);parents.More();parents.Next())
        if(!closure.contains(parents.Value().get())&&!affected.contains(parents.Value().get()))
          throw std::runtime_error("Exact STEP shared geometry has an unproved external context reference.");
    }
    // Keep the parsed model's actual WorkSession alive. A temporary session
    // destructor can clear its model even when other handles retain it.
    primary.Reader->WS()->ComputeGraph(true);
    // ReadStream initialized the transfer process before the context-local
    // item closures were installed. Recreate that private process against the
    // final graph so every added entity has a valid transfer slot.
    primary.Reader->WS()->InitTransferReader(5);
    primary.Reader->SetSystemLengthUnit(reader.SystemLengthUnit());
    const int roots=reader.NbRootsForTransfer();
    if(primary.Reader->NbRootsForTransfer()!=roots)throw std::runtime_error("Exact STEP context copy changed transfer root membership.");
    for(int index=1;index<=roots;++index)
      if(primary.Counterpart(reader.RootForTransfer(index))!=primary.Reader->RootForTransfer(index))
        throw std::runtime_error("Exact STEP context copy changed an actual transfer root identity or order.");
    return std::move(primary.Reader);
  }

OcctSharp_ExactStepOffsetReadScope::OcctSharp_ExactStepOffsetReadScope(STEPControl_Reader& reader)
{
  auto model=reader.StepModel();
  if(model.IsNull())throw std::runtime_error("Exact STEP units require a loaded reader model.");
  std::vector<occ::handle<StepGeom_OffsetCurve3d>> offsets;
  for(int index=1;index<=model->NbEntities();++index)
  {
    const auto offset=occ::down_cast<StepGeom_OffsetCurve3d>(model->Value(index));
    if(!offset.IsNull())offsets.push_back(offset);
  }
  if(offsets.empty())return;
  const Interface_Graph graph(model);
  std::vector<ContextEntry> representations;bool conflict=false;
  std::vector<occ::handle<StepGeom_OffsetCurve3d>> represented;
  for(const auto& offset:offsets)
  {
    const auto contexts=RepresentationContexts(graph,offset,reader.SystemLengthUnit());
    if(contexts.empty())
    {
      // Part21 may retain unused geometric entities. Never guess their units
      // or modify them. A selected bare root depending on such an offset still
      // requires a proved representation context and is explicitly rejected.
      for(int index=1;index<=reader.NbRootsForTransfer();++index)
      {
        const auto root=reader.RootForTransfer(index);
        if(root==offset)throw std::runtime_error("Exact STEP units have no reachable geometry representation.");
        for(auto parents=graph.TypedSharings(offset,root->DynamicType());parents.More();parents.Next())
          if(parents.Value()==root)throw std::runtime_error("Exact STEP units have no reachable geometry representation.");
      }
      continue;
    }
    represented.push_back(offset);
    for(const auto& entry:contexts)
    {
      ReadOffsetDistance(offset,entry.second); // Validate all distances before activation.
      if(!SameFactors(contexts.front().second,entry.second))conflict=true;
      if(std::none_of(representations.begin(),representations.end(),[&](const auto& value){return value.first==entry.first;}))
        representations.push_back(entry);
    }
  }
  offsets.swap(represented);
  if(offsets.empty())return;
  std::unique_ptr<STEPControl_Reader> prepared;
  if(conflict)
  {
    prepared=ContextLocalReader(reader,representations);model=prepared->StepModel();
    prepared->SetShapeFixParameters(reader.GetShapeFixParameters());
    const auto process=reader.GetShapeProcessFlags();if(process.second)prepared->SetShapeProcessFlags(process.first);
    offsets.clear();
    const Interface_Graph privateGraph(model);
    for(int index=1;index<=model->NbEntities();++index)
    {
      const auto offset=occ::down_cast<StepGeom_OffsetCurve3d>(model->Value(index));
      if(!offset.IsNull()&&privateGraph.TypedSharings(offset,STANDARD_TYPE(StepRepr_Representation)).NbEntities()!=0)offsets.push_back(offset);
    }
  }
  const Interface_Graph transferGraph(model);
  std::vector<double> adjusted;adjusted.reserve(offsets.size());myOriginals.reserve(offsets.size());
  for(const auto& offset:offsets)
  {
    const auto factors=RepresentationFactors(transferGraph,offset,reader.SystemLengthUnit());
    adjusted.push_back(ReadOffsetDistance(offset,factors));myOriginals.emplace_back(offset,offset->Distance());
  }
  if(prepared){mySavedReader=std::make_unique<STEPControl_Reader>(reader);myReader=&reader;}
  for(size_t index=0;index<myOriginals.size();++index)myOriginals[index].first->SetDistance(adjusted[index]);
  if(prepared)reader=*prepared;
}
OcctSharp_ExactStepOffsetReadScope::~OcctSharp_ExactStepOffsetReadScope()
{
  for(const auto& original:myOriginals)original.first->SetDistance(original.second);
  if(mySavedReader)*myReader=*mySavedReader;
}

  class ExactPlaneCoordinates
  {
    gp_Ax3 myPlane;
    static double Roundoff(const gp_Pnt& point,const gp_Pnt& origin)
    {
      const double scale=1+std::max({std::abs(point.X()),std::abs(point.Y()),std::abs(point.Z()),
        std::abs(origin.X()),std::abs(origin.Y()),std::abs(origin.Z())});
      return 256*std::numeric_limits<double>::epsilon()*scale;
    }
  public:
    explicit ExactPlaneCoordinates(const gp_Ax3& plane):myPlane(plane){}
    gp_Pnt2d Point(const gp_Pnt& point) const
    {
      const gp_Vec delta(myPlane.Location(),point);
      const double normal=delta.Dot(gp_Vec(myPlane.Direction()));
      if(!std::isfinite(normal)||std::abs(normal)>Roundoff(point,myPlane.Location()))
        throw std::runtime_error("Exact planar pcurve basis has a nonplanar control point.");
      return gp_Pnt2d(delta.Dot(gp_Vec(myPlane.XDirection())),delta.Dot(gp_Vec(myPlane.YDirection())));
    }
    gp_Dir2d Direction(const gp_Dir& direction) const
    {
      if(std::abs(direction.Dot(myPlane.Direction()))>256*std::numeric_limits<double>::epsilon())
        throw std::runtime_error("Exact planar pcurve basis has a transverse direction.");
      return gp_Dir2d(direction.Dot(myPlane.XDirection()),direction.Dot(myPlane.YDirection()));
    }
    double OffsetSign(const gp_Dir& direction) const
    {
      // Ax3 can be indirect. The 2D offset's Z is X cross Y, not necessarily
      // Ax3.Direction(). Keeping both axes also preserves mirrored conics.
      const gp_Vec normal=gp_Vec(myPlane.XDirection()).Crossed(gp_Vec(myPlane.YDirection()));
      const double sign=gp_Vec(direction).Dot(normal);
      if(!std::isfinite(sign)||std::abs(std::abs(sign)-1)>256*std::numeric_limits<double>::epsilon()
        ||gp_Vec(direction).Crossed(normal).Magnitude()>256*std::numeric_limits<double>::epsilon())
        throw std::runtime_error("Exact planar pcurve offset direction is not parallel to its plane normal.");
      return sign<0?-1:1;
    }
  };
  occ::handle<Geom2d_Curve> ExactPlanarCurve(const occ::handle<Geom_Curve>& curve,
    const ExactPlaneCoordinates& plane,const int depth=0)
  {
    if(curve.IsNull()||depth>32)throw std::runtime_error("Invalid or over-depth exact planar pcurve basis.");
    const auto offset=occ::down_cast<Geom_OffsetCurve>(curve);
    if(!offset.IsNull())
    {
      const double sign=plane.OffsetSign(offset->Direction());
      if(!std::isfinite(offset->Offset()))throw std::runtime_error("Nonfinite exact planar offset.");
      return new Geom2d_OffsetCurve(ExactPlanarCurve(offset->BasisCurve(),plane,depth+1),offset->Offset()*sign);
    }
    const auto trimmed=occ::down_cast<Geom_TrimmedCurve>(curve);
    if(!trimmed.IsNull())return new Geom2d_TrimmedCurve(ExactPlanarCurve(trimmed->BasisCurve(),plane,depth+1),
      trimmed->FirstParameter(),trimmed->LastParameter(),true,false);
    const auto line=occ::down_cast<Geom_Line>(curve);
    if(!line.IsNull())return new Geom2d_Line(plane.Point(line->Position().Location()),plane.Direction(line->Position().Direction()));
    const auto circle=occ::down_cast<Geom_Circle>(curve);
    if(!circle.IsNull())return new Geom2d_Circle(gp_Ax22d(plane.Point(circle->Location()),
      plane.Direction(circle->Position().XDirection()),plane.Direction(circle->Position().YDirection())),circle->Radius());
    const auto ellipse=occ::down_cast<Geom_Ellipse>(curve);
    if(!ellipse.IsNull())return new Geom2d_Ellipse(gp_Ax22d(plane.Point(ellipse->Location()),
      plane.Direction(ellipse->Position().XDirection()),plane.Direction(ellipse->Position().YDirection())),ellipse->MajorRadius(),ellipse->MinorRadius());
    const auto spline=occ::down_cast<Geom_BSplineCurve>(curve);
    if(!spline.IsNull())
    {
      if(spline->NbPoles()>100000||spline->NbKnots()>100000)throw std::runtime_error("Exact planar spline control budget exceeded.");
      NCollection_Array1<gp_Pnt2d> poles(1,spline->NbPoles());
      for(int i=1;i<=spline->NbPoles();++i)poles(i)=plane.Point(spline->Pole(i));
      // Every control point is proved coplanar, so this is an affine change of
      // coordinates of the whole rational curve, not a sampled approximation.
      if(spline->IsRational())return new Geom2d_BSplineCurve(poles,*spline->Weights(),spline->Knots(),
        spline->Multiplicities(),spline->Degree(),spline->IsPeriodic());
      return new Geom2d_BSplineCurve(poles,spline->Knots(),spline->Multiplicities(),spline->Degree(),spline->IsPeriodic());
    }
    const auto bezier=occ::down_cast<Geom_BezierCurve>(curve);
    if(!bezier.IsNull())
    {
      NCollection_Array1<gp_Pnt2d> poles(1,bezier->NbPoles());
      for(int i=1;i<=bezier->NbPoles();++i)poles(i)=plane.Point(bezier->Pole(i));
      if(bezier->IsRational())return new Geom2d_BezierCurve(poles,*bezier->Weights());
      return new Geom2d_BezierCurve(poles);
    }
    throw std::runtime_error("Exact planar pcurve basis kind is unsupported; no fitted fallback is allowed.");
  }
}
int32_t OcctSharp_CompleteImportedExactPlanarOffsetPcurves(const TopoDS_Shape& shape)
{
  struct Change{TopoDS_Edge edge;TopoDS_Face face;occ::handle<Geom2d_Curve> curve;double tolerance;};
  std::vector<Change> changes;
  NCollection_IndexedMap<TopoDS_Shape,TopTools_ShapeMapHasher> faces;TopExp::MapShapes(shape,TopAbs_FACE,faces);
  if(faces.Extent()>100000)throw std::runtime_error("Exact planar pcurve face budget exceeded.");
  for(int i=1;i<=faces.Extent();++i)
  {
    const auto face=TopoDS::Face(faces(i));auto surface=BRep_Tool::Surface(face);
    const auto trimmed=occ::down_cast<Geom_RectangularTrimmedSurface>(surface);
    if(!trimmed.IsNull())surface=trimmed->BasisSurface();
    const auto plane=occ::down_cast<Geom_Plane>(surface);if(plane.IsNull())continue;
    NCollection_IndexedMap<TopoDS_Shape,TopTools_ShapeMapHasher> edges;TopExp::MapShapes(face,TopAbs_EDGE,edges);
    if(edges.Extent()>100000)throw std::runtime_error("Exact planar pcurve edge budget exceeded.");
    for(int j=1;j<=edges.Extent();++j)
    {
      const auto edge=TopoDS::Edge(edges(j));double first,last;const auto curve=BRep_Tool::Curve(edge,first,last);
      if(curve.IsNull()||!HasOffset(curve))continue;
      double lower,upper;const auto existing=BRep_Tool::CurveOnSurface(edge,face,lower,upper);
      if(existing.IsNull()||first!=lower||last!=upper||!BRep_Tool::SameRange(edge)||!BRep_Tool::SameParameter(edge))
        throw std::runtime_error("Exact planar pcurve requires the actual edge-face parameter contract.");
      changes.push_back({edge,face,ExactPlanarCurve(curve,ExactPlaneCoordinates(plane->Position())),BRep_Tool::Tolerance(edge)});
    }
  }
  // All geometry proofs and curve allocations precede the first local BRep edit.
  // Actual edge-face membership supplies identity, never distance matching.
  BRep_Builder builder;
  for(const auto& change:changes)builder.UpdateEdge(change.edge,change.curve,change.face,change.tolerance);
  return static_cast<int32_t>(changes.size());
}

int32_t OcctSharp_CompleteImportedExactPlanarOffsetPcurves(const occ::handle<TDocStd_Document>& document)
{
  if(document.IsNull())throw std::runtime_error("Exact planar pcurve import document is null.");
  const auto tool=XCAFDoc_DocumentTool::ShapeTool(document->Main());
  NCollection_Sequence<TDF_Label> roots;tool->GetFreeShapes(roots);int32_t repaired=0;
  for(const auto& root:roots)repaired+=OcctSharp_CompleteImportedExactPlanarOffsetPcurves(tool->GetShape(root));
  return repaired;
}

  occ::handle<Geom_Curve> TransferBasis(const occ::handle<Geom_Curve>& curve,const int depth=0)
  {
    if(curve.IsNull()||depth>32)throw std::runtime_error("Invalid or over-depth exact STEP transfer skeleton.");
    const auto offset=occ::down_cast<Geom_OffsetCurve>(curve);
    if(!offset.IsNull())return TransferBasis(offset->BasisCurve(),depth+1);
    const auto trimmed=occ::down_cast<Geom_TrimmedCurve>(curve);
    if(!trimmed.IsNull()&&HasOffset(trimmed->BasisCurve()))return TransferBasis(trimmed->BasisCurve(),depth+1);
    return curve;
  }
  occ::handle<Geom2d_Curve> AnalyticTransferPcurve(const occ::handle<Geom2d_Curve>& curve,const int depth=0)
  {
    if(curve.IsNull()||depth>32)throw std::runtime_error("Invalid or over-depth exact STEP parameter skeleton.");
    const auto trimmed=occ::down_cast<Geom2d_TrimmedCurve>(curve);
    if(!trimmed.IsNull())
    {
      const auto converted=AnalyticTransferPcurve(trimmed->BasisCurve(),depth+1);
      if(converted==trimmed->BasisCurve())return curve;
      return new Geom2d_TrimmedCurve(converted,trimmed->FirstParameter(),trimmed->LastParameter(),true,false);
    }
    const auto offset=occ::down_cast<Geom2d_OffsetCurve>(curve);
    if(offset.IsNull())return curve;
    const auto basis=AnalyticTransferPcurve(offset->BasisCurve(),depth+1);
    const auto circle=occ::down_cast<Geom2d_Circle>(basis);
    if(!circle.IsNull())
    {
      auto axes=circle->Position();
      const double sign=axes.XDirection().Crossed(axes.YDirection())>0?1:-1;
      const double radius=circle->Radius()+sign*offset->Offset();
      if(!std::isfinite(radius)||radius==0)return curve;
      if(radius<0)axes=gp_Ax22d(axes.Location(),axes.XDirection().Reversed(),axes.YDirection().Reversed());
      // This is the exact circular offset identity with the SAME parameter,
      // axes/sense and period. Only the private writer skeleton is changed.
      return new Geom2d_Circle(axes,std::abs(radius));
    }
    const auto line=occ::down_cast<Geom2d_Line>(basis);
    if(!line.IsNull())
    {
      const auto axis=line->Position();const auto direction=axis.Direction();
      const gp_Vec2d normal(direction.Y(),-direction.X());
      return new Geom2d_Line(gp_Ax2d(axis.Location().Translated(normal*offset->Offset()),direction));
    }
    return curve; // No approximate or fitted fallback for other basis kinds.
  }
  void PrepareTransferSkeleton(const TopoDS_Shape& copy)
  {
    // The skeleton is private, never returned or written. It lets the stock
    // converter allocate supported STEP entity types and real topology binders.
    // CompleteExactStepOffsets replaces every altered geometry before Write.
    BRep_Builder builder;
    NCollection_IndexedMap<TopoDS_Shape,TopTools_ShapeMapHasher> edges,faces;
    TopExp::MapShapes(copy,TopAbs_EDGE,edges);TopExp::MapShapes(copy,TopAbs_FACE,faces);
    if(edges.Extent()>100000||faces.Extent()>100000)throw std::runtime_error("Exact STEP transfer skeleton topology budget exceeded.");
    for(int i=1;i<=edges.Extent();++i)
    {
      const auto edge=TopoDS::Edge(edges(i));double first,last;TopLoc_Location location;
      const auto curve=BRep_Tool::Curve(edge,location,first,last);
      if(curve.IsNull()||!HasOffset(curve))continue;
      builder.UpdateEdge(edge,TransferBasis(curve),location,BRep_Tool::Tolerance(edge));
      builder.Range(edge,first,last,true);
    }
    for(int i=1;i<=faces.Extent();++i)
    {
      const auto face=TopoDS::Face(faces(i));
      NCollection_IndexedMap<TopoDS_Shape,TopTools_ShapeMapHasher> faceEdges;TopExp::MapShapes(face,TopAbs_EDGE,faceEdges);
      for(int index=1;index<=faceEdges.Extent();++index)
      {
        const auto edge=TopoDS::Edge(faceEdges(index));double first,last;
        const auto original=BRep_Tool::CurveOnSurface(edge,face,first,last);
        if(original.IsNull())continue;
        const auto converted=AnalyticTransferPcurve(original);
        if(converted==original)continue;
        if(BRep_Tool::IsClosed(edge,face))throw std::runtime_error("Exact analytic parameter skeleton needs an explicit seam pair mapping.");
        builder.UpdateEdge(edge,converted,face,BRep_Tool::Tolerance(edge));builder.Range(edge,face,first,last);
      }
      TopLoc_Location location;auto surface=BRep_Tool::Surface(face,location);
      const auto trimmed=occ::down_cast<Geom_RectangularTrimmedSurface>(surface);
      if(!trimmed.IsNull())surface=trimmed->BasisSurface();
      const auto extrusion=occ::down_cast<Geom_SurfaceOfLinearExtrusion>(surface);
      if(extrusion.IsNull()||!HasOffset(extrusion->BasisCurve()))continue;
      builder.UpdateFace(face,new Geom_SurfaceOfLinearExtrusion(TransferBasis(extrusion->BasisCurve()),extrusion->Direction()),
        location,BRep_Tool::Tolerance(face));
    }
  }
}
bool OcctSharp_TransferExactStepOffsets(STEPControl_Writer& writer,const TopoDS_Shape& original)
{
  NCollection_IndexedMap<TopoDS_Shape,TopTools_ShapeMapHasher> edges;TopExp::MapShapes(original,TopAbs_EDGE,edges);
  if(edges.Extent()>100000)throw std::runtime_error("Exact STEP private transfer edge budget exceeded.");
  bool hasOffset=false;
  for(int i=1;i<=edges.Extent();++i)
  {
    double first,last;const auto curve=BRep_Tool::Curve(TopoDS::Edge(edges(i)),first,last);
    if(!curve.IsNull()&&HasOffset(curve)){hasOffset=true;break;}
  }
  if(!hasOffset)return false;
  BRepBuilderAPI_Copy privateCopy(original,true,false);
  if(!privateCopy.IsDone()||privateCopy.Shape().IsNull())throw std::runtime_error("Exact STEP private geometry copy failed.");
  PrepareTransferSkeleton(privateCopy.Shape());
  writer.SetShapeProcessFlags(ShapeProcess::OperationsFlags{});
  if(writer.Transfer(privateCopy.Shape(),STEPControl_AsIs)!=IFSelect_RetDone)
    throw std::runtime_error("Exact STEP private transfer failed.");
  CompleteExactStepOffsets(writer,original,&privateCopy);
  return true;
}

  occ::handle<TDocStd_Document> PrivateFullXdeTree(
    const occ::handle<TDocStd_Document>& source,const occ::handle<TDocStd_Application>& application)
  {
    if(source->ToReferencesNumber()!=0)
      throw std::runtime_error("Exact XDE private export cannot yet clone external document references.");
    const auto labels=DocumentLabels(source);const auto signature=AttributeSignature(source);
    const auto links=TreeLinkSignature(source);
    occ::handle<TDF_DataSet> data=new TDF_DataSet;
    for(const auto& label:labels)
    {
      data->AddLabel(label);
      for(TDF_AttributeIterator it(label);it.More();it.Next())
        if(it.Value()->ID()!=TDocStd_Owner::GetID())data->AddAttribute(it.Value());
    }
    data->AddRoot(labels.front());
    for(NCollection_Map<occ::handle<TDF_Attribute>>::Iterator it(data->Attributes());it.More();it.Next())
    {
      occ::handle<TDF_DataSet> references=new TDF_DataSet;it.Key()->References(references);
      for(NCollection_Map<TDF_Label>::Iterator ref(references->Labels());ref.More();ref.Next())
        if(!data->ContainsLabel(ref.Key()))throw std::runtime_error("Exact XDE private export has an external attribute label reference.");
      for(NCollection_Map<occ::handle<TDF_Attribute>>::Iterator ref(references->Attributes());ref.More();ref.Next())
        if(!data->ContainsAttribute(ref.Key()))throw std::runtime_error("Exact XDE private export has an external attribute reference.");
    }
    occ::handle<TDocStd_Document> treeCopy;application->NewDocument(TCollection_ExtendedString("BinXCAF"),treeCopy);
    if(treeCopy.IsNull())throw std::runtime_error("Exact XDE private tree creation failed.");
    try
    {
      occ::handle<TDF_RelocationTable> relocation=new TDF_RelocationTable(false);
      // Typed HasRelocation() leaves its output unchanged on a missing/null
      // input in 8.0.1. TreeNode::Paste reuses that output for all four links.
      // AfterRelocate makes absent links explicitly null once the complete
      // attribute relocation map has been prepared by CopyTool.
      relocation->AfterRelocate(true);
      relocation->SetRelocation(labels.front(),treeCopy->GetData()->Root());TDF_CopyTool::Copy(data,relocation);
      if(AttributeSignature(treeCopy)!=signature)throw std::runtime_error("Exact XDE full attribute relocation lost a label/type/ID.");
      if(TreeLinkSignature(treeCopy)!=links)throw std::runtime_error("Exact XDE full relocation changed a tree link.");
      std::stringstream bytes(std::ios::in|std::ios::out|std::ios::binary);
      if(application->SaveAs(treeCopy,bytes)!=PCDM_SS_OK)throw std::runtime_error("Exact XDE private binary serialization failed.");
      if(bytes.tellp()>512*1024*1024)throw std::runtime_error("Exact XDE private binary memory budget exceeded.");
      bytes.seekg(0);occ::handle<TDocStd_Document> isolated;
      if(application->Open(bytes,isolated)!=PCDM_RS_OK||isolated.IsNull())throw std::runtime_error("Exact XDE private binary reload failed.");
      // Recreate only explicitly copied empty label paths that persistence may
      // omit. Missing authored attributes remain a hard failure below.
      for(const auto& label:labels)
      {
        TDF_Label target;TDF_Tool::Label(isolated->GetData(),LabelEntry(label).c_str(),target,true);
        occ::handle<XCAFDoc_ShapeMapTool> originalCache;
        if(label.FindAttribute(XCAFDoc_ShapeMapTool::GetID(),originalCache))
        {
          // This non-persistent attribute is a derived subshape lookup cache.
          // Rebuild it from the actual reloaded shape on its relocated label;
          // copying its original map would reintroduce source TShape sharing.
          const auto shape=XCAFDoc_ShapeTool::GetShape(target);
          if(shape.IsNull())throw std::runtime_error("Exact XDE shape-map cache label has no reloaded shape.");
          XCAFDoc_ShapeMapTool::Set(target)->SetShape(shape);
        }
      }
      if(AttributeSignature(isolated)!=signature)
      {
        const auto reloaded=AttributeSignature(isolated);std::vector<std::string> missing,extra;
        std::set_difference(signature.begin(),signature.end(),reloaded.begin(),reloaded.end(),std::back_inserter(missing));
        std::set_difference(reloaded.begin(),reloaded.end(),signature.begin(),signature.end(),std::back_inserter(extra));
        std::string message="Exact XDE binary driver attribute mismatch.";
        for(const auto& item:missing)message+=" Missing["+item+"]";
        for(const auto& item:extra)message+=" Extra["+item+"]";
        application->Close(isolated);throw std::runtime_error(message);
      }
      if(TreeLinkSignature(isolated)!=links)
      {
        application->Close(isolated);throw std::runtime_error("Exact XDE binary reload changed a tree link.");
      }
      application->Close(treeCopy);return isolated;
    }
    catch(...){application->Close(treeCopy);throw;}
  }
}
OcctSharp_ExactStepXdeWriteScope::OcctSharp_ExactStepXdeWriteScope(const occ::handle<TDocStd_Document>& source)
{
  if(source.IsNull())throw std::runtime_error("Exact XDE write source document is null.");
  const auto authorGeometry=DocumentNamedShapes(source);
  bool hasOffsets=false;for(const auto& shape:authorGeometry)if(GeometryHasOffsets(shape)){hasOffsets=true;break;}
  if(!hasOffsets){myDocument=source;return;}
  myApplication=new TDocStd_Application;OcctSharp::Native::ConfigureSafeNamedDataDrivers(myApplication);
  try
  {
    myDocument=PrivateFullXdeTree(source,myApplication);
    const auto privateGeometry=DocumentGeometry(myDocument);
    NCollection_IndexedMap<TopoDS_Shape,TopTools_ShapeMapHasher> authorShapes,privateShapes;
    for(const auto& shape:authorGeometry)TopExp::MapShapes(shape,authorShapes);
    TopExp::MapShapes(privateGeometry,privateShapes);
    std::unordered_set<const TopoDS_TShape*> authorTShapes;
    for(int i=1;i<=authorShapes.Extent();++i)authorTShapes.insert(authorShapes(i).TShape().get());
    for(int i=1;i<=privateShapes.Extent();++i)
      if(authorTShapes.contains(privateShapes(i).TShape().get()))throw std::runtime_error("Exact XDE binary clone still shares author TShape geometry.");
    myExactGeometry=std::make_unique<BRepBuilderAPI_Copy>(privateGeometry,true,false);
    if(!myExactGeometry->IsDone())throw std::runtime_error("Exact XDE geometry correspondence copy failed.");
    PrepareTransferSkeleton(privateGeometry);
  }
  catch(...)
  {
    if(!myDocument.IsNull())myApplication->Close(myDocument);
    myDocument.Nullify();throw;
  }
}

  if(!myApplication.IsNull()&&!myDocument.IsNull())
    try{myApplication->Close(myDocument);}catch(const Standard_Failure&){}
}

OcctSharp_ExactStepOffsetReport OcctSharp_ExactStepXdeWriteScope::Complete(STEPControl_Writer& writer) const
{
  if(!HasExactOffsets())return OcctSharp_CompleteTransferredExactStepOffsets(writer);
  const auto process=writer.WS()->TransferWriter()->FinderProcess();
  if(process.IsNull())throw std::runtime_error("Exact XDE completion has no actual transfer process.");
  TopoDS_Compound sources;BRep_Builder builder;builder.MakeCompound(sources);
  NCollection_IndexedMap<TopoDS_Shape,TopTools_ShapeMapHasher> mapped;
  for(int i=1;i<=process->NbMapped();++i)
  {
    const auto mapper=occ::down_cast<TransferBRep_ShapeMapper>(process->Mapped(i));if(mapper.IsNull())continue;
    const auto& shape=mapper->Value();if(shape.IsNull()||(shape.ShapeType()!=TopAbs_EDGE&&shape.ShapeType()!=TopAbs_FACE)||mapped.Contains(shape))continue;
    mapped.Add(shape);builder.Add(sources,shape);
  }
  return CompleteExactStepOffsets(writer,sources,myExactGeometry.get(),true);
}
