// A cache of unique labels alone cannot enforce a path-depth or expanded-visit
// limit: a shared definition may later be reached through a longer/doubled path.
// Cache complete subtree summaries and add them at every incoming occurrence.
inline void ValidatePaths(const NCollection_Sequence<TDF_Label>& roots,
                          const occ::handle<TDF_Data>& data) {
  struct Summary { int height=0; int occurrences=0; };
  NCollection_DataMap<TDF_Label,Summary> cache;
  NCollection_Map<TDF_Label> active;
  std::function<Summary(const TDF_Label&)> visit=[&](const TDF_Label& label)->Summary {
    if(label.IsNull()||label.Data()!=data)throw std::invalid_argument("Shared assembly path crosses its exact source document.");
    if(active.Size()>16)throw std::invalid_argument("Expanded shared path depth exceeds16 before traversal.");
    Summary summary;
    if(cache.Find(label,summary)) {
      if(active.Size()+summary.height>16)throw std::invalid_argument("A shared cached subtree exceeds the complete path depth16.");
      return summary;
    }
    if(!active.Add(label))throw std::invalid_argument("Shared assembly path has a cycle.");
    if(active.Size()+cache.Size()>4096)throw std::invalid_argument("Shared assembly unique label budget exceeded.");
    if(XCAFDoc_ShapeTool::IsComponent(label)) {
      TDF_Label referred;
      if(!XCAFDoc_ShapeTool::GetReferredShape(label,referred))throw std::invalid_argument("Shared component has no definition.");
      summary=visit(referred);++summary.height;++summary.occurrences;
    } else if(XCAFDoc_ShapeTool::IsAssembly(label)) {
      NCollection_Sequence<TDF_Label> children;XCAFDoc_ShapeTool::GetComponents(label,children,false);
      for(const auto& child:children) {
        const auto descendant=visit(child);
        summary.height=std::max(summary.height,descendant.height+1);
        summary.occurrences+=descendant.occurrences;
        if(summary.occurrences>4096)throw std::invalid_argument("Expanded shared component paths exceed4096.");
      }
    }
    if(summary.height>16||summary.occurrences>4096)throw std::invalid_argument("Expanded shared path depth/occurrence budget exceeded.");
    active.Remove(label);cache.Bind(label,summary);return summary;
  };
  int total=0;for(const auto& root:roots) {
    total+=visit(root).occurrences;
    if(total>4096)throw std::invalid_argument("Selected roots exceed the combined expanded occurrence budget.");
  }
}
// Canonicalize each assembly once, retaining the shared exact label clone map.
// Its own placement is pushed onto outgoing components: L(def) * L(component).
// Nested definitions do the same independently. Consequently every leaf receives
// L(root) * L(c1) * L(def1) * ... * L(cN), with no extra wrapper or duplicate L.
inline void Normalize(const occ::handle<XCAFDoc_ShapeTool>& source,
                      const occ::handle<XCAFDoc_ShapeTool>& destination,
                      const NCollection_DataMap<TDF_Label,TDF_Label>& labels) {
  for(NCollection_DataMap<TDF_Label,TDF_Label>::Iterator item(labels);item.More();item.Next()) {
    if(item.Key().Data()!=source->Label().Data()||item.Value().Data()!=destination->Label().Data())
      throw std::invalid_argument("Assembly clone map is outside its exact source/destination document.");
    if(!XCAFDoc_ShapeTool::IsAssembly(item.Key()))continue;
    const auto placement=XCAFDoc_ShapeTool::GetLocation(item.Key());
    const auto shape=XCAFDoc_ShapeTool::GetShape(item.Key());
    if(shape.IsNull()||!shape.Location().IsEqual(placement))
      throw std::invalid_argument("Assembly location attribute and definition shape disagree.");
    if(!XCAFDoc_ShapeTool::GetLocation(item.Value()).IsIdentity())
      throw std::invalid_argument("Private clone assembly is not canonical identity.");
    const auto& trsf=placement.Transformation();
    if(!placement.IsIdentity()&&std::abs(trsf.ScaleFactor()-1.)>1e-12)
      throw std::invalid_argument("Located assembly definitions require rigid positive placement.");
    for(int row=1;row<=3;++row)for(int col=1;col<=4;++col)
      if(!std::isfinite(trsf.Value(row,col))||std::abs(trsf.Value(row,col))>1e12)
        throw std::invalid_argument("Assembly definition placement is nonfinite/excessive.");
    NCollection_Sequence<TDF_Label> components;
    XCAFDoc_ShapeTool::GetComponents(item.Key(),components,false);
    if(components.Length()==0||components.Length()>4096)
      throw std::invalid_argument("Empty/large normalized assembly.");
    for(const auto& component:components) {
      TDF_Label clone;
      occ::handle<XCAFDoc_Location> location;
      if(!labels.Find(component,clone)||!component.FindAttribute(XCAFDoc_Location::GetID(),location)||location.IsNull())
        throw std::invalid_argument("Missing exact component clone/location.");
      const auto normalized=placement*location->Get();
      const auto& matrix=normalized.Transformation();
      for(int row=1;row<=3;++row)for(int col=1;col<=4;++col)
        if(!std::isfinite(matrix.Value(row,col))||std::abs(matrix.Value(row,col))>1e12)
          throw std::invalid_argument("Normalized component placement is nonfinite/excessive.");
      XCAFDoc_Location::Set(clone,normalized);
    }
  }
  // The source is read only. Destination topology is rebuilt from canonical labels
  // after the caller's independent geometry/metadata copies, by UpdateAssemblies.
}
